//<![CDATA[
  let activeCategoryFilter = 'All';

  document.addEventListener('DOMContentLoaded', function() {
    const initLang = localStorage.getItem('acadtrack-lang') || 'en';
    setLanguage(initLang);
    
    executeFeaturesFiltering();
  });

  function setLanguage(lang) {
    localStorage.setItem('acadtrack-lang', lang);
    document.documentElement.lang = lang;
    const isRTL = lang === 'ar';
    const container = document.querySelector('.saas-container');
    if (container) container.dir = isRTL ? 'rtl' : 'ltr';

    document.querySelectorAll('[data-lang]').forEach(el => {
      el.style.display = el.getAttribute('data-lang') === lang ? '' : 'none';
    });
  }

  // Categories Selector trigger
  window.filterFeatures = function(category, tabId) {
    activeCategoryFilter = category;
    
    document.querySelectorAll('.features-tab').forEach(t => t.classList.remove('active'));
    const activeTab = document.getElementById(tabId);
    if (activeTab) activeTab.classList.add('active');

    executeFeaturesFiltering();
  };

  function executeFeaturesFiltering() {
    const cards = document.querySelectorAll('.feature-card');
    cards.forEach(card => {
      const cat = card.getAttribute('data-feat-cat');
      
      let matches = false;
      if (activeCategoryFilter === 'All') {
        matches = true;
      } else if (activeCategoryFilter === 'Future') {
        matches = card.classList.contains('future');
      } else {
        matches = cat === activeCategoryFilter && card.classList.contains('deployed');
      }

      card.style.display = matches ? 'flex' : 'none';
    });
  }
//]]>
</script>
