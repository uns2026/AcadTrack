<script>
  (function() {
    // Paste your Google Apps Script Web App Deployment URL here:
    const GOOGLE_APPS_SCRIPT_URL = "https://script.google.com/macros/s/AKfycbzcKp49sXa43kg0oBDkM0QfbxzoJ_KyZ2sXByqnaRn4Z7_SSfehQpTa9_2-FWjWv_0nSg/exec";

    function initContactPage() {
      // Auto-detect and inherit user language selection
      const initLang = localStorage.getItem('acadtrack-lang') || 'en';
      setLanguage(initLang);
    }

    function setLanguage(lang) {
      localStorage.setItem('acadtrack-lang', lang);
      document.documentElement.lang = lang;
      const isRTL = lang === 'ar';
      const container = document.querySelector('.saas-container');
      if (container) container.dir = isRTL ? 'rtl' : 'ltr';

      document.querySelectorAll('[data-lang]').forEach(el => {
        el.style.display = el.getAttribute('data-lang') === lang ? '' : 'none';
      });
    }

    window.handleFormSubmit = function() {
      const form = document.getElementById('contact-form');
      const submitBtn = document.getElementById('submit-btn');
      const btnIcon = document.getElementById('btn-icon');
      const btnText = document.getElementById('btn-text');

      if (!form || !submitBtn || !btnIcon || !btnText) return;

      // Ensure Apps Script URL is updated
      if (GOOGLE_APPS_SCRIPT_URL.includes("YOUR_DEPLOYED_URL")) {
        alert("Configuration Error: Please paste your active Google Apps Script web app URL inside the script block.");
        return;
      }

      // Enter loading state
      submitBtn.disabled = true;
      btnIcon.className = "fa-solid fa-spinner";
      
      const currentLang = localStorage.getItem('acadtrack-lang') || 'en';
      btnText.innerHTML = currentLang === 'ar' ? 'جاري الإرسال...' : (currentLang === 'fr' ? 'Envoi...' : 'Sending...');

      // Convert Form inputs into url-encoded payloads to completely bypass CORS preflight blocks
      const formData = new FormData(form);
      const urlEncodedPayload = new URLSearchParams(formData);

      fetch(GOOGLE_APPS_SCRIPT_URL, {
        method: "POST",
        body: urlEncodedPayload,
        headers: {
          "Content-Type": "application/x-www-form-urlencoded"
        }
      })
      .then(res => res.json())
      .then(data => {
        if (data && data.status === "success") {
          // Success Response
          triggerToast(
            currentLang === 'ar' ? "تم الإرسال بنجاح" : "Success",
            currentLang === 'ar' ? "تم استلام رسالتك وتأكيد استلام بريد إلكتروني على حسابك." : "Your message has been sent successfully and a receipt confirmation has been delivered to your email.",
            "success"
          );
          form.reset();
        } else {
          // Internal Apps Script execution failure
          triggerToast(
            currentLang === 'ar' ? "فشل الإرسال" : "Error",
            data.message || "An error occurred while compiling your message.",
            "error"
          );
        }
      })
      .catch(err => {
        // System connection error
        triggerToast(
          currentLang === 'ar' ? "خطأ في الشبكة" : "Network Error",
          currentLang === 'ar' ? "يرجى التحقق من اتصالك بالإنترنت وإعادة المحاولة." : "Could not establish connection to delivery gateway.",
          "error"
        );
      })
      .finally(() => {
        // Restore submit button states
        submitBtn.disabled = false;
        btnIcon.className = "fa-solid fa-paper-plane";
        btnText.innerHTML = `
          <span data-lang="en">Send Message</span>
          <span data-lang="fr">Envoyer</span>
          <span data-lang="ar">إرسال الرسالة</span>
        `;
        setLanguage(currentLang);
      });
    };

    function triggerToast(title, desc, type) {
      const container = document.getElementById('toast-portal');
      if (!container) return;

      const toastId = 'toast_' + Date.now();
      const isSuccess = type === 'success';
      const iconClass = isSuccess ? 'fa-circle-check' : 'fa-triangle-exclamation';

      const toastHTML = `
        <div class="saas-toast ${type}" id="${toastId}">
          <i class="fa-solid ${iconClass}"></i>
          <div class="toast-body">
            <div class="toast-title">${title}</div>
            <div class="toast-desc">${desc}</div>
          </div>
        </div>
      `;

      container.innerHTML += toastHTML;

      // Animate entry
      setTimeout(() => {
        const el = document.getElementById(toastId);
        if (el) el.classList.add('active');
      }, 50);

      // Auto Dismiss after 4 seconds
      setTimeout(() => {
        const el = document.getElementById(toastId);
        if (el) {
          el.classList.remove('active');
          setTimeout(() => el.remove(), 350);
        }
      }, 4000);
    }

    // Initialize page behaviors
    initContactPage();
  })();
</script>
