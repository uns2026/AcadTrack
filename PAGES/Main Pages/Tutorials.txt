<script>
//<![CDATA[
  let activeSearchQuery = '';

  document.addEventListener('DOMContentLoaded', function() {
    // MOVE MODAL DIRECTLY TO BODY ROOT
    // This detaches the modal from Blogger/Layout wrappers to guarantee it floats over sticky headers
    const modal = document.getElementById('tutorialFullscreenModal');
    if (modal) {
      document.body.appendChild(modal);
    }

    // Sync active language selection
    const activeLang = localStorage.getItem('acadtrack-lang') || 'en';
    setLanguage(activeLang);

    // Click handler to load the dynamically selected guide
    document.querySelectorAll('.tutorial-card').forEach(card => {
      card.addEventListener('click', function(e) {
        e.preventDefault();
        const url = this.getAttribute('data-url');
        openTutorialModal(url);
      });
    });

    // Handle Escape keyboard press to trigger safe modal exit
    window.addEventListener('keydown', function(e) {
      if (e.key === 'Escape') {
        closeTutorialModal();
      }
    });
  });

  // Language rendering controller
  function setLanguage(lang) {
    localStorage.setItem('acadtrack-lang', lang);
    document.documentElement.lang = lang;
    const isRTL = lang === 'ar';
    
    // Set layout direction on the primary container
    const saasBox = document.querySelector('.saas-container');
    if (saasBox) saasBox.dir = isRTL ? 'rtl' : 'ltr';

    // Synchronize modal direction and structural classes even after moving to document body
    const modal = document.getElementById('tutorialFullscreenModal');
    if (modal) {
      modal.dir = isRTL ? 'rtl' : 'ltr';
      if (isRTL) {
        modal.classList.add('lang-ar');
      } else {
        modal.classList.remove('lang-ar');
      }
    }

    document.querySelectorAll('[data-lang]').forEach(el => {
      el.style.display = el.getAttribute('data-lang') === lang ? '' : 'none';
    });
  }

  // Filter list items through query selection
  window.filterSaaSTutorials = function(query) {
    activeSearchQuery = query.toLowerCase().trim();
    let matchCount = 0;
    const cards = document.querySelectorAll('.tutorial-card');

    cards.forEach(card => {
      const cardText = card.textContent.toLowerCase();
      const matchesSearch = !activeSearchQuery || cardText.includes(activeSearchQuery);

      if (matchesSearch) {
        card.style.display = 'flex';
        matchCount++;
      } else {
        card.style.display = 'none';
      }
    });

    const emptyBox = document.getElementById('searchEmptyState');
    if (emptyBox) {
      emptyBox.style.display = matchCount === 0 ? 'block' : 'none';
    }
  };

  // Open & inject URL into DOM frame
  function openTutorialModal(url) {
    const modal = document.getElementById('tutorialFullscreenModal');
    const iframe = document.getElementById('modalActiveIFrame');

    if (!modal || !iframe) return;

    iframe.src = url; 
    modal.classList.add('active');
    document.body.style.overflow = 'hidden'; // Lock webpage scrollbars
  }

  // Close & reset iframe src to prevent background execution processes
  window.closeTutorialModal = function() {
    const modal = document.getElementById('tutorialFullscreenModal');
    const iframe = document.getElementById('modalActiveIFrame');

    if (!modal || !iframe) return;

    modal.classList.remove('active');
    iframe.src = 'about:blank'; // Free browser allocation memory 
    document.body.style.overflow = ''; // Unlock webpage scrollbars
  };
//]]>
</script>
