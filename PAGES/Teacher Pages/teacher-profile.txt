  window.saveProfileSecurity = function() {
    if (!activeUserObj) return;

    const oldPass = document.getElementById('sec-input-old-password').value;
    const newPass = document.getElementById('sec-input-password').value;
    const confirmPass = document.getElementById('sec-input-confirm-password').value;

    let decryptedCurrent = '';
    try {
      decryptedCurrent = activeUserObj.password ? atob(activeUserObj.password) : '123';
    } catch(e) {
      decryptedCurrent = activeUserObj.password || '123';
    }

    // 1. Old password verify check
    if (oldPass !== decryptedCurrent) {
      const msgErr = {
        en: "Verification Failed: Incorrect old password.",
        fr: "Vérification Échouée : Ancien mot de passe incorrect.",
        ar: "فشل التحقق: كلمة المرور الحالية غير صحيحة."
      };
      const lang = localStorage.getItem('acadtrack-lang') || 'en';
      alert(msgErr[lang] || msgErr.en);
      return;
    }

    // 2. Length check
    if (!newPass || newPass.length < 6) {
      const msgErr = {
        en: "Security Error: New password must be at least 6 characters.",
        fr: "Erreur Sécurité : Le mot de passe doit comporter au moins 6 caractères.",
        ar: "خطأ أمني: يجب أن تتكون كلمة المرور الجديدة من 6 أحرف على الأقل."
      };
      const lang = localStorage.getItem('acadtrack-lang') || 'en';
      alert(msgErr[lang] || msgErr.en);
      return;
    }

    // 3. Confirm match check
    if (newPass !== confirmPass) {
      const msgErr = {
        en: "Security Error: Confirm password does not match new password.",
        fr: "Erreur Sécurité : La confirmation ne correspond pas au nouveau mot de passe.",
        ar: "خطأ أمني: تأكيد كلمة المرور غير متطابق مع كلمة المرور الجديدة."
      };
      const lang = localStorage.getItem('acadtrack-lang') || 'en';
      alert(msgErr[lang] || msgErr.en);
      return;
    }

    // Securely write using standard Base64 encoding parameters
    activeUserObj.password = btoa(newPass);

    updateUserDatabaseRecord(activeUserObj);

    document.getElementById('profile-security-form').reset();
    switchProfileSubpanel('feed');
    triggerSuccessToast("Password Saved", "Security credentials successfully updated.");
  };

  // Sync session and local database collections
  function updateUserDatabaseRecord(updatedUserObj) {
    localStorage.setItem('acadtrack_currentUser', JSON.stringify(updatedUserObj));

    let localUsers = [];
    try {
      const stored = localStorage.getItem('acadtrack_users');
      if (stored) localUsers = JSON.parse(stored);
    } catch(err) { localUsers = []; }

    const index = localUsers.findIndex(u => u.uid === updatedUserObj.uid);
    if (index !== -1) {
      localUsers[index] = updatedUserObj;
      localStorage.setItem('acadtrack_users', JSON.stringify(localUsers));
    }

    // Sync general header parameters
    localStorage.setItem('teacherName', updatedUserObj.name);
    localStorage.setItem('teacherAvatar', updatedUserObj.avatar);
  }
