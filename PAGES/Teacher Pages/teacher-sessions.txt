  function updateSessionStats() {
    const todayStr = '2026-05-25'; // Simulated clock baseline
    const completed = sessionsCache.filter(s => s.date < todayStr).length;
    const upcoming = sessionsCache.filter(s => s.date >= todayStr).length;
    const tdCount = sessionsCache.filter(s => s.type === 'TD').length;

    const setText = (id, val) => { const el = document.getElementById(id); if(el) el.textContent = val; };
    setText('sess-stat-total', sessionsCache.length);
    setText('sess-stat-completed', completed);
    setText('sess-stat-upcoming', upcoming);
    setText('sess-stat-td', tdCount);
  }

  function renderRosterSessionsTable() {
    const tbody = document.querySelector('#sessions-table tbody');
    if (!tbody) return;

    const query = activeSessFilter.query.toLowerCase().trim();
    const todayStr = '2026-05-25';

    let filtered = sessionsCache.filter(s => {
      const matchesQuery = !query || s.module.toLowerCase().includes(query) || s.group.toLowerCase().includes(query) || s.room.toLowerCase().includes(query);
      const matchesModule = activeSessFilter.module === 'All' || s.module === activeSessFilter.module;
      const matchesType = activeSessFilter.type === 'All' || s.type === activeSessFilter.type;
      
      let matchesStatus = true;
      if (activeSessFilter.status === 'Upcoming') {
        matchesStatus = s.date >= todayStr;
      } else if (activeSessFilter.status === 'Completed') {
        matchesStatus = s.date < todayStr;
      }
      
      return matchesQuery && matchesModule && matchesType && matchesStatus;
    });

    // Execute Sort
    filtered.sort((a, b) => {
      const dateTimeA = new Date(`${a.date}T${a.time}`);
      const dateTimeB = new Date(`${b.date}T${b.time}`);
      return activeSessFilter.sortBy === 'date-asc' ? dateTimeA - dateTimeB : dateTimeB - dateTimeA;
    });

    if (filtered.length === 0) {
      tbody.innerHTML = `<tr><td colspan="7" style="text-align:center;color:var(--text-light);"><span data-lang="en">No planned sessions found</span><span data-lang="fr">Aucune séance planifiée</span><span data-lang="ar">لا توجد حصص مجدولة</span></td></tr>`;
      translateSaaSElements(tbody);
      return;
    }

    tbody.innerHTML = filtered.map(sess => {
      const typeClass = sess.type.toLowerCase();
      const isPast = sess.date < todayStr;
      
      const attendanceActionBtn = !isPast ? `
        <button class="btn-saas btn-saas-primary" onclick="startRosterQuickRollCall('${escapeHtml(sess.module)}', '${escapeHtml(sess.group)}', '${sess.date}')">
          <i class="fa-solid fa-clipboard-user"></i> <span data-lang="en">Take Attendance</span><span data-lang="fr">Faire l'Appel</span><span data-lang="ar">بدء تسجيل الحضور</span>
        </button>
      ` : `
        <button class="btn-saas btn-saas-secondary" onclick="startRosterQuickRollCall('${escapeHtml(sess.module)}', '${escapeHtml(sess.group)}', '${sess.date}')" style="background:#ecfdf5; color:#047857; border-color:#a7f3d0;">
          <i class="fa-solid fa-user-check"></i> <span data-lang="en">Edit Sheet</span><span data-lang="fr">Modifier Fiche</span><span data-lang="ar">تعديل كشف الغياب</span>
        </button>
      `;

      return `
        <tr>
          <td style="font-weight:700; color:var(--text);">${escapeHtml(sess.module)}</td>
          <td>${escapeHtml(sess.group)}</td>
          <td><span class="badge badge-${typeClass}">${sess.type}</span></td>
          <td>${sess.date} @ ${sess.time}</td>
          <td>${escapeHtml(sess.room)}</td>
          <td style="text-align:center;">${attendanceActionBtn}</td>
          <td class="action-col" style="text-align:center; display: flex; gap: 4px; justify-content: center;">
            <button class="btn-saas btn-saas-secondary btn-icon-only" onclick="editSessionNode('${sess.id}')" title="Edit"><i class="fa-solid fa-pen"></i></button>
            <button class="btn-saas btn-saas-danger btn-icon-only" onclick="promptDeleteSessionNode('${sess.id}')" title="Delete"><i class="fa-solid fa-trash"></i></button>
          </td>
        </tr>
      `;
    }).join('');

    translateSaaSElements(tbody);
  }

  function saveSessionFromForm() {
    const getVal = id => { const el = document.getElementById(id); return el ? el.value.trim() : ''; };

    const module = getVal('sess-input-module');
    const group = getVal('sess-input-group');
    const type = getVal('sess-input-type');
    const room = getVal('sess-input-room');
    const date = getVal('sess-input-date');
    const time = getVal('sess-input-time');

    if (!module || !group || !type || !room || !date || !time) return;

    const payload = { id: editingId || 'sess_' + Date.now(), module, group, type, date, time, room };

    if (editingId) {
      const idx = sessionsCache.findIndex(s => s.id === editingId);
      if (idx !== -1) sessionsCache[idx] = payload;
    } else {
      sessionsCache.push(payload);
    }

    window.saveData('sessions', sessionsCache);
    loadSaaSDatabases();
    renderRosterSessionsTable();
    updateSessionStats();
    closeScheduleSessionModal();
  }

      if (status) {

        const node = { studentId: std.id, status, date, module, method: 'Manual', timestamp: new Date().toISOString() };

        else attendanceCache.push(node);
      }

    loadSaaSDatabases();
    updateAttendanceDashboardMetrics();
    renderCarouselCard();

      'Attendance Saved',
      'Attendance sheet submitted successfully.',
      '#10b981'
    );
  };

      const hrs = String(Math.floor(secLeft / 3600)).padStart(2, '0');
      const mins = String(Math.floor((secLeft % 3600) / 60)).padStart(2, '0');
      const secs = String(secLeft % 60).padStart(2, '0');
      const formatted = `Valid for ${hrs}:${mins}:${secs}`;
      if (timer) timer.textContent = `Valid for ${hrs}:${mins}:${secs}`;
      const fsTimer = document.getElementById('qr-fs-timer');
      if (fsTimer) fsTimer.textContent = `Valid for ${hrs}:${mins}:${secs}`;

      secLeft--;
      updateTimerText();
      if (secLeft <= 0) {
        clearInterval(qrTimerInterval);
        if (timer) timer.textContent = "QR Code Expired. Please regenerate.";
        if (qrFrame) { qrFrame.style.opacity = '0.4'; qrFrame.style.borderColor = '#ef4444'; }

  function interpolatePath(points, stepsPerSegment = 10) {
    const interpolated = [];
    for (let i = 0; i < points.length - 1; i++) {
      const start = points[i];
      const end = points[i + 1];
      for (let step = 0; step < stepsPerSegment; step++) {
        const pct = step / stepsPerSegment;
        const lat = start[0] + (end[0] - start[0]) * pct;
        const lon = start[1] + (end[1] - start[1]) * pct;
        interpolated.push([lat, lon]);
      }
    }
    interpolated.push(points[points.length - 1]);
    return interpolated;
  }

  function getDistance(lat1, lon1, lat2, lon2) {
    const R = 6371000;
    const dLat = (lat2 - lat1) * Math.PI / 180;
    const dLon = (lon2 - lon1) * Math.PI / 180;
    const a = Math.sin(dLat/2) * Math.sin(dLat/2) +
              Math.cos(lat1 * Math.PI / 180) * Math.cos(lat2 * Math.PI / 180) *
              Math.sin(dLon/2) * Math.sin(dLon/2);
    const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a));
    return R * c;
  }

      btn.innerHTML = `<i class="fa-solid fa-square-stop"></i> <span data-lang="en">${stopLabel.en}</span><span data-lang="fr">${stopLabel.fr}</span><span data-lang="ar">${stopLabel.ar}</span>`;
      btn.style.background = '#dc2626'; btn.style.borderColor = '#dc2626';
    }
    translateSaaSElements(btn);

    let queue = [...activeCohortStudents];
    activeDemoInterval = setInterval(() => {
      if (queue.length === 0) {
        clearInterval(activeDemoInterval);
        activeDemoInterval = null;
        if (btn) {
          btn.innerHTML = `<i class="fa-solid fa-play"></i> <span data-lang="en">${launchLabel.en}</span><span data-lang="fr">${launchLabel.fr}</span><span data-lang="ar">${launchLabel.ar}</span>`;
          btn.style.background = '#7c3aed'; btn.style.borderColor = '#7c3aed';
        }
        translateSaaSElements(btn);
        return;
      }

      const randIdx = Math.floor(Math.random() * queue.length);
      const targetStudent = queue.splice(randIdx, 1)[0];
      const methods = ['QR', 'NFC Card', 'GPS'];
      const chosenMethod = methods[Math.floor(Math.random() * methods.length)];
      const randStatus = Math.random() > 0.9 ? 'L' : 'P';
      
      const module = document.getElementById('att-select-session').value;
      const date = document.getElementById('att-select-date').value;

      const idx = attendanceCache.findIndex(a => a.studentId === targetStudent.id && a.date === date && a.module === module);
      const payload = { studentId: targetStudent.id, status: randStatus, date, module, method: chosenMethod, timestamp: new Date().toISOString() };

      if (idx !== -1) attendanceCache[idx] = payload;
      else attendanceCache.push(payload);

      window.saveData('attendance', attendanceCache);
      loadSaaSDatabases();
      carouselAttendanceDraft[targetStudent.id] = randStatus;

      updateAttendanceDashboardMetrics();
      if (currentCarouselIndex < activeCohortStudents.length) updateCarouselCardContent();

      triggerTopRightToastNotification(targetStudent, randStatus, chosenMethod);
    }, Math.floor(Math.random() * 2000) + 1500);
  };

  function triggerTopRightToastNotification(student, status, method) {
    const container = document.getElementById('toastPortalContainer');
    if (!container) return;

    const currentLang = localStorage.getItem('acadtrack-lang') || 'en';
    const toastId = 'toast_' + Date.now();

    const statusLabels = { P: { en: "Present", fr: "Présent", ar: "حاضر" }, L: { en: "Late", fr: "En retard", ar: "متأخر" } };
    const methodLabels = {
      'QR': { en: 'QR Code', fr: 'Scanner QR', ar: 'تلقائياً عبر الـ QR' },
      'NFC Card': { en: 'Student Card', fr: 'Carte NFC', ar: 'تلقائياً عبر بطاقة الطالب' },
      'GPS': { en: 'GPS Geofence', fr: 'Vérification GPS', ar: 'تلقائياً عبر موقع الـ GPS' }
    };

    const title = statusLabels[status][currentLang];
    const methodLabel = methodLabels[method][currentLang];

    const bodyText = {
      en: `${student.firstName} ${student.lastName} just checked-in.`,
      fr: `${student.firstName} ${student.lastName} s'est signalé.`,
      ar: `تم تسجيل دخول الطالب ${student.firstName} ${student.lastName} بنجاح.`
    };

    container.innerHTML += `
      <div class="saas-toast-alert" id="${toastId}">
        <div class="saas-toast-icon" style="background:${status === 'P' ? '#ecfdf5' : '#fffbeb'}; color:${status === 'P' ? '#10b981' : '#d97706'}">
          <i class="fa-solid ${status === 'P' ? 'fa-check' : 'fa-clock'}"></i>
        </div>
        <div class="saas-toast-body">
          <div class="saas-toast-title">${title} — <span style="font-weight:500; font-size:11px; color:#64748b;">${methodLabel}</span></div>
          <div class="saas-toast-desc">${bodyText[currentLang]}</div>
        </div>
      </div>
    `;

    setTimeout(() => { const el = document.getElementById(toastId); if (el) el.classList.add('active'); }, 50);
    setTimeout(() => { const el = document.getElementById(toastId); if (el) { el.classList.remove('active'); setTimeout(() => el.remove(), 350); } }, 4500);
  }

  function triggerToastNotification(iconHTML, title, description, color) {
    const overlay = document.getElementById('att-toast-overlay');
    const iconWrap = document.getElementById('att-toast-icon');
    const titleEl = document.getElementById('att-toast-title');
    const descEl = document.getElementById('att-toast-desc');

    if (overlay && iconWrap && titleEl && descEl) {
      iconWrap.innerHTML = iconHTML;
      iconWrap.style.color = color;
      titleEl.textContent = title;
      descEl.textContent = description;

      overlay.classList.add('active');
      setTimeout(() => overlay.classList.remove('active'), 2500);
    }
  }

  function renderTrendsSummaryTable() {
    const tbody = document.getElementById('trends-risk-table-body');
    if (!tbody) return;

    const riskCohort = studentsCache.filter(s => {
      const rate = parseFloat(s.attendanceRate || '100');
      return rate < 80;
    });

    if (riskCohort.length === 0) {
      tbody.innerHTML = `<tr><td colspan="4" style="text-align:center;color:#94a3b8;"><span data-lang="en">All students are above the 80% warning threshold.</span><span data-lang="fr">Tous au-dessus de 80%.</span><span data-lang="ar">جميع الطلبة فوق حد الإنذار البالغ 80%</span></td></tr>`;
      translateSaaSElements(tbody);
      return;
    }

    tbody.innerHTML = riskCohort.map(std => `
      <tr>
        <td style="font-family:var(--font-mono);">${std.id}</td>
        <td style="font-weight:700; color:#0f172a;">${escapeHtml(std.lastName)} ${escapeHtml(std.firstName)}</td>
        <td>${std.group}</td>
        <td style="font-weight:700; color:#dc2626;">${std.attendanceRate}</td>
      </tr>
    `).join('');

    translateSaaSElements(tbody);
  }

  function translateSaaSElements(container) {
    if (!container) return;
    const currentLang = localStorage.getItem('acadtrack-lang') || 'en';
    container.querySelectorAll('[data-lang]').forEach(el => {
      el.style.display = el.getAttribute('data-lang') === currentLang ? '' : 'none';
    });
  }

  function escapeHtml(text) {
    if (text == null) return '';
    const div = document.createElement('div');
    div.textContent = String(text);
    return div.innerHTML;
  }
