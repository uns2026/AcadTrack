          <i class="fa-regular fa-comments" style="font-size: 3rem; margin-bottom: 10px; display: block; color: #cbd5e1;"></i>

          <p style="font-size: 12px; margin-top: 4px;"><span data-lang="en">Correspond with colleagues or academic coordinators.</span><span data-lang="fr">Échangez avec vos étudiants ou l'administration.</span><span data-lang="ar">تواصل مع الزملاء أو رؤساء الأقسام أو طلبتك مباشرة.</span></p>
        </div>
      </div>
    `;
    translateMessagesPanelElements();
  }

  function loadMessagesDatabase() {
    messagesDatabase = window.getData('messages');
  }

  function renderContactsDirectory() {
    const wrapper = document.getElementById('contactsRosterWrapper');
    if (!wrapper) return;

    const q = searchQuery.toLowerCase().trim();

    const filtered = contactsRoster.filter(c => {
      const matchesQuery = !q || c.name.toLowerCase().includes(q) || c.role.toLowerCase().includes(q);
      const matchesCategory = activeCategoryFilter === 'All' || c.type === activeCategoryFilter;
      return matchesQuery && matchesCategory;
    });

    wrapper.innerHTML = filtered.map(c => {
      const chatLogs = messagesDatabase.filter(m => m.contactId === c.id);
      const lastMsg = chatLogs[chatLogs.length - 1];
      const snippet = lastMsg ? lastMsg.text : 'No messages exchange.';
      const time = lastMsg ? lastMsg.date : '';
      const isUnread = chatLogs.some(m => m.sender === 'received' && m.unread);
      
      const activeClass = c.id === activeContactId ? 'active' : '';
      const unreadClass = isUnread ? 'unread' : '';

      return `
        <div class="msg-contact-item ${activeClass} ${unreadClass}" onclick="selectSaaSTargetContact('${c.id}')">
          <img src="${c.avatar}" class="msg-avatar" alt="${c.name}" />
          <div class="msg-contact-meta">
            <div class="msg-contact-name">${escapeHtml(c.name)}</div>
            <div class="msg-contact-role">
              <span>${escapeHtml(c.role)}</span>
              <span style="font-family:var(--font-mono); font-size:10px; color:#cbd5e1;">${time}</span>
            </div>
            <div class="msg-last-snip">${escapeHtml(snippet)}</div>
          </div>
        </div>
      `;
    }).join('');
  }

  window.filterSaaSContactsList = function(val) {
    searchQuery = val;
    renderContactsDirectory();
  };

  window.filterContactCategory = function(cat) {
    activeCategoryFilter = cat;
    document.querySelectorAll('.msg-cat-tab').forEach(t => t.classList.remove('active'));
    
    const tabId = cat === 'All' ? 'tabCatAll' : (cat === 'Admin' ? 'tabCatAdmin' : (cat === 'Teacher' ? 'tabCatTeacher' : 'tabCatStudent'));
    const tabEl = document.getElementById(tabId);
    if (tabEl) tabEl.classList.add('active');

    renderContactsDirectory();
    translateMessagesPanelElements();
  };

  window.selectSaaSTargetContact = function(id) {
    activeContactId = id;
    
    // Mark all incoming messages for selected contact as read
    messagesDatabase.forEach(m => {
      if (m.contactId === id && m.sender === 'received') m.unread = false;
    });
    window.saveData('messages', messagesDatabase);

    // Re-evaluate roster active visual selections
    renderContactsDirectory();

    const contactObj = contactsRoster.find(c => c.id === id);
    if (!contactObj) return;

    // Update Chat Header Workspace
    document.getElementById('activeChatName').textContent = contactObj.name;
    document.getElementById('activeChatStatus').style.display = 'inline';
    document.getElementById('chat-input-bar-form').style.display = 'flex';

    // Toggle dynamic mobile Messenger active view class
    const portal = document.getElementById('msgPortalContainer');
    if (portal) portal.classList.add('chat-active');

    renderActiveConversationView();
  };

  // Native Messenger back navigation on mobile
  window.backToInbox = function() {
    activeContactId = null;
    
    const portal = document.getElementById('msgPortalContainer');
    if (portal) portal.classList.remove('chat-active');
    
    renderContactsDirectory();
  };

  function renderActiveConversationView() {
    const body = document.getElementById('chatRosterBody');
    if (!body) return;

    const chatLogs = messagesDatabase.filter(m => m.contactId === activeContactId);

    if (chatLogs.length === 0) {
      body.innerHTML = `
        <div class="pt-center" style="height: 100%; display: flex; align-items: center; justify-content: center;">
          <div style="text-align: center; color: var(--text-light); font-size: 13px; font-weight: 600;">
            <span data-lang="en">No message logs. Start typing below...</span>
            <span data-lang="fr">Aucun message. Saisissez votre texte ci-dessous...</span>
            <span data-lang="ar">لا توجد سجلات رسائل بينكما. اكتب رسالة أدناه للبدء...</span>
          </div>
        </div>
      `;
      translateMessagesPanelElements();
      return;
    }

    // Build chat log view and append typing indicator holder
    body.innerHTML = chatLogs.map(m => {
      const wrapClass = m.sender === 'sent' ? 'sent' : 'received';
      const senderLabel = m.sender === 'sent' ? 'Me' : contactsRoster.find(c => c.id === m.contactId)?.name || 'Incoming';
      
      return `
        <div class="msg-bubble-wrap ${wrapClass}">
          <div class="msg-bubble">
            <span class="chat-sender">${escapeHtml(senderLabel)}</span>
            <span>${escapeHtml(m.text)}</span>
          </div>
          <span class="msg-timestamp">${m.date}</span>
        </div>
      `;
    }).join('') + `
      <!-- Animated Typing indicator skeleton block -->
      <div class="typing-indicator-wrap" id="typingIndicator">
        <div class="typing-bubble">
          <div class="typing-dot"></div>
          <div class="typing-dot"></div>
          <div class="typing-dot"></div>
        </div>
        <span class="msg-timestamp" style="margin-left: 4px; align-self: flex-start;"><span data-lang="en">Typing...</span><span data-lang="fr">Écrit...</span><span data-lang="ar">يكتب الآن...</span></span>
      </div>
    `;

    body.scrollTop = body.scrollHeight; // Force scroll to bottom
  }

  // Submit and store manual message
  window.submitLiveSaaSMessage = function() {
    const input = document.getElementById('chatViewportInputField');
    if (!input || !activeContactId) return;

    const text = input.value.trim();
    if (!text) return;

    const timeStr = new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });

    const newMsg = {
      contactId: activeContactId,
      sender: 'sent',
      text: text,
      date: timeStr,
      unread: false
    };

    messagesDatabase.push(newMsg);
    window.saveData('messages', messagesDatabase);

    input.value = ''; // Reset
    renderActiveConversationView();
    renderContactsDirectory();

    // Trigger automatic sandbox simulated replying delay (1.5 seconds)
    simulateAdminBypassReply(activeContactId);
  };

        "الله يحفظك أستاذة زكية راني مع صحابي نوجدوا في العرض وبإذن الله نبعثوه في الوقت المحدد",
        "صحيتي أستاذة زكية يكثر خيرك رانا نتبعو في الحصص ونعوضو في الغيابات اللي فاتونا"
      ]
    };

    const pool = responsePool[targetContact.type] || responsePool['Admin'];
    const chosenReply = pool[Math.floor(Math.random() * pool.length)];

    // Show Typing indicator block
    const indicator = document.getElementById('typingIndicator');
    const body = document.getElementById('chatRosterBody');
    if (indicator) {
      indicator.classList.add('active');
      translateMessagesPanelElements();
      if (body) body.scrollTop = body.scrollHeight;
    }

    setTimeout(() => {
      // Hide typing loader
      if (indicator) indicator.classList.remove('active');

      const replyTime = new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });
      
      const replyMsg = {
        contactId: contactId,
        sender: 'received',
        text: chosenReply,
        date: replyTime,
        unread: true
      };

      messagesDatabase.push(replyMsg);
      window.saveData('messages', messagesDatabase);

      if (activeContactId === contactId) {
        renderActiveConversationView();
        replyMsg.unread = false;
        window.saveData('messages', messagesDatabase);
      }

      renderContactsDirectory();

    }, 1800); // 1.8 seconds processing time
  }

  function translateMessagesPanelElements() {
    const currentLang = localStorage.getItem('acadtrack-lang') || 'en';
    document.querySelectorAll('#panel-messages [data-lang]').forEach(el => {
      el.style.display = el.getAttribute('data-lang') === currentLang ? '' : 'none';
    });
  }

  function escapeHtml(text) {
    if (text == null) return '';
    const div = document.createElement('div');
    div.textContent = String(text);
    return div.innerHTML;
  }
