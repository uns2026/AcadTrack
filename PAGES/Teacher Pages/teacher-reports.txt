  window.compileSaaSReport = function() {
    const module = document.getElementById('rep-input-module').value;
    const group = document.getElementById('rep-input-group').value;
    const students = window.getData('students') || [];
    const cohort = students.filter(s => s.group === group);

    if (cohort.length === 0) {
      const currentLang = localStorage.getItem('acadtrack-lang') || 'en';
      const msg = {
        en: "This student group does not have any active roster entries. Compilation bypassed.",
        fr: "Ce groupe n'a aucun étudiant inscrit. Édition annulée.",
        ar: "الفوج المحدد فارغ ولا يحتوي على طلبة مسجلين. تم إلغاء التوليد."
      };
      alert(msg[currentLang] || msg.en);
      return;
    }

    const reportRef = 'REP_' + Date.now().toString().slice(-5);
    const newReport = {
      id: reportRef,
      scope: selectedTemplate,
      group: group,
      module: module,
      format: selectedFormat.toUpperCase(),
      date: new Date().toLocaleString()
    };

    reportsCache.push(newReport);
    window.saveData('reports', reportsCache);

    if (selectedFormat === 'csv' || selectedFormat === 'excel') {
      compileCSVOrExcelFile(cohort, module, group, selectedFormat);
    } else if (selectedFormat === 'pdf') {
      compilePDFPrintLayout(cohort, module, group);
    }

    renderReportsLogs();
    
    triggerToastNotification(
      '<i class="fa-solid fa-circle-check"></i>',
      'Report Compiled',
      `Successfully generated report ${reportRef}.`,
      '#10b981'
    );
  };

  function compileCSVOrExcelFile(cohort, module, group, format) {
    let fileContent = '';
    
    if (selectedTemplate === 'Attendance Registry') {
      fileContent = "Student ID,Student Name,Group,Module,Attendance Rate\n";
      cohort.forEach(s => {
        fileContent += `${s.id},"${s.lastName} ${s.firstName}","${s.group}","${module}",${s.attendanceRate || '100%'}\n`;
      });
    } else if (selectedTemplate === 'Evaluations Summary') {
      fileContent = "Student ID,Student Name,Group,Module,Exam Grade,TD Grade,TP Grade,Assignment,Participation,Average GPA,GPA Award\n";
      const grades = window.getData('grades') || [];
      cohort.forEach(s => {
        const g = grades.find(gr => gr.studentId === s.id && gr.module === module) || { exam: '—', td: '—', tp: '—', assignment: '—', participation: '—' };
        const roundedAvg = calculateRosterGradeAverage(g.exam, g.td, g.tp, g.assignment, g.participation);
        const letter = roundedAvg >= 10 ? 'Pass' : 'Fail';
        fileContent += `${s.id},"${s.lastName} ${s.firstName}","${s.group}","${module}",${g.exam},${g.td},${g.tp},${g.assignment},${g.participation},${roundedAvg},${letter}\n`;
      });
    } else {
      fileContent = "Metrics Summary,Value\n";
      fileContent += "Target Module Name," + module + "\n";
      fileContent += "Target Class Group," + group + "\n";
      fileContent += "Active Cohort Size," + cohort.length + "\n";
      fileContent += "Attendance Average,88%\n";
      fileContent += "Pass Rate (GPA >= 10),94.2%\n";
    }

    const extension = format === 'csv' ? 'csv' : 'xls';
    const fileBlob = new Blob([fileContent], { type: format === 'csv' ? 'text/csv' : 'application/vnd.ms-excel' });
    
    const link = document.createElement('a');
    link.href = URL.createObjectURL(fileBlob);
    link.download = `report_${group.toLowerCase().replace(/\s+/g,'_')}.${extension}`;
    link.click();
  }

  function calculateRosterGradeAverage(exam, td, tp, assignment, participation) {
    const ex = parseFloat(exam || '0');
    const t_d = parseFloat(td || '0');
    const t_p = parseFloat(tp || '0');
    const ass = parseFloat(assignment || '0');
    const part = parseFloat(participation || '0');

    const average = (ex * 0.50) + (t_d * 0.20) + (t_p * 0.15) + (ass * 0.10) + (part * 0.05);
    return parseFloat(average.toFixed(2));
  }

  window.downloadSaaSReportArchive = function(refId) {
    const rep = reportsCache.find(r => r.id === refId);
    if (!rep) return;

    const students = window.getData('students') || [];
    const cohort = students.filter(s => s.group === rep.group);

    if (rep.format === 'PDF') {
      compilePDFPrintLayout(cohort, rep.module, rep.group);
    } else {
      compileCSVOrExcelFile(cohort, rep.module, rep.group, rep.format.toLowerCase());
    }
  };

  function triggerToastNotification(iconHTML, title, description, color) {
    const overlay = document.getElementById('att-toast-overlay');
    const iconWrap = document.getElementById('att-toast-icon');
    const titleEl = document.getElementById('att-toast-title');
    const descEl = document.getElementById('att-toast-desc');

    if (overlay && iconWrap && titleEl && descEl) {
      iconWrap.innerHTML = iconHTML;
      iconWrap.style.color = color;
      titleEl.textContent = title;
      descEl.textContent = description;

      overlay.classList.add('active');
      setTimeout(() => overlay.classList.remove('active'), 2500);
    }
  }

  function translateSaaSElements(container) {
    if (!container) return;
    const currentLang = localStorage.getItem('acadtrack-lang') || 'en';
    container.querySelectorAll('[data-lang]').forEach(el => {
      el.style.display = el.getAttribute('data-lang') === currentLang ? '' : 'none';
    });
  }

  function escapeHtml(text) {
    if (text == null) return '';
    const div = document.createElement('div');
    div.textContent = String(text);
    return div.innerHTML;
  }
