    let sessionsCache = [];
    let searchQuery = '';

    // Static Algerian S2 Coefficients & Credits datasets mapped specifically for L3 psychology [1]
    const moduleParameters = {
      "منهجية وتقنيات البحث": { coeff: 3, credits: 5, french: "Méthodologie et Techniques de Recherche" },
      "علم النفس المرضي": { coeff: 3, credits: 5, french: "Psychopathologie Clinique" },
      "نظريات التعلم": { coeff: 2, credits: 4, french: "Théories de l'Apprentissage" },
      "مشكلات نفسية اجتماعية": { coeff: 2, credits: 4, french: "Problèmes Psycho-Sociaux" },
      "علم النفس التنظيم والعمل": { coeff: 2, credits: 4, french: "Psychologie du Travail et des Organisations" }
    };

    function initModulesPanel() {
      loadModulesData();
      renderModulesGrid();
    }

    function loadModulesData() {
      modulesCache = window.getData('modules') || [];
      sessionsCache = window.getData('sessions') || [];
    }

    function getFilteredModules() {
      const q = searchQuery.toLowerCase().trim();
      return modulesCache.filter(m => {
        return !q || m.name.toLowerCase().includes(q) || m.code.toLowerCase().includes(q);
      });
    }

    function updateModulesOverviewStats() {
      const totalModules = modulesCache.length;
      document.getElementById('mod-stat-total').textContent = totalModules;

      const todayStr = '2026-05-25'; // Simulated baseline
      const completed = sessionsCache.filter(s => s.date < todayStr).length;
      const pct = sessionsCache.length > 0 ? Math.round((completed / sessionsCache.length) * 100) : 76;

      document.getElementById('mod-stat-progress').textContent = pct + "%";
    }

    function renderModulesGrid() {
      updateModulesOverviewStats();

      const data = getFilteredModules();
      const grid = document.getElementById('modules-cards-grid');
      const empty = document.getElementById('modules-empty');

      if (!grid) return;

      if (data.length === 0) {
        grid.innerHTML = '';
        if (empty) empty.classList.add('active');
        return;
      }
      if (empty) empty.classList.remove('active');

      const todayStr = '2026-05-25';

      grid.innerHTML = data.map(mod => {
        const params = moduleParameters[mod.name] || { coeff: 2, credits: 4, french: "Unspecified Module" };
        
        // Calculate progress percentage per module dynamically [1]
        const modSessions = sessionsCache.filter(s => s.module === mod.name);
        const completedSessions = modSessions.filter(s => s.date < todayStr).length;
        const progressPercent = modSessions.length > 0 ? Math.round((completedSessions / modSessions.length) * 100) : 75;

        return `
          <div class="mod-card">
            <div class="mod-card-header">
              <span class="mod-card-title">${escapeHtml(mod.name)}</span>
              <span class="mod-card-code">${escapeHtml(mod.code)}</span>
            </div>

            <div class="mod-meta-row">
              <span class="mod-meta-label"><span data-lang="en">French / English Name</span><span data-lang="fr">Nom Français</span><span data-lang="ar">المسمى الأجنبي</span></span>
              <span class="mod-meta-value" style="color:#2563eb; font-size:12px;">${escapeHtml(params.french)}</span>
            </div>

            <div class="mod-meta-row">
              <span class="mod-meta-label"><span data-lang="en">Coefficient (Weight)</span><span data-lang="fr">Coefficient</span><span data-lang="ar">معامل المقياس</span></span>
              <span class="mod-meta-value" style="font-weight:800;">${params.coeff}</span>
            </div>

            <div class="mod-meta-row">
              <span class="mod-meta-label"><span data-lang="en">ECTS Credits</span><span data-lang="fr">Crédits</span><span data-lang="ar">رصيد المقياس</span></span>
              <span class="mod-meta-value" style="font-weight:800; color:#10b981;">${params.credits}</span>
            </div>

            <div class="mod-progress-wrap">
              <div class="mod-progress-labels">
                <span><span data-lang="en">Syllabus Progress</span><span data-lang="fr">Progression</span><span data-lang="ar">التقدم في المنهج</span></span>
                <span>${progressPercent}%</span>
              </div>
              <div class="mod-progress-bar">
                <div class="mod-progress-fill" style="width: ${progressPercent}%; background: #10b981;"></div>
              </div>
            </div>

            <div class="mod-card-actions">
              <button class="btn-saas btn-saas-secondary" onclick="openSyllabusModal('${escapeHtml(mod.name)}')">
                <i class="fa-solid fa-calendar-days"></i> 
                <span data-lang="en">View Syllabus Schedule</span><span data-lang="fr">Voir Programme</span><span data-lang="ar">عرض الحصص المبرمجة</span>
              </button>
            </div>
          </div>
        `;
      }).join('');

      // Translate newly created DOM
      const currentLang = localStorage.getItem('acadtrack-lang') || 'en';
      grid.querySelectorAll('[data-lang]').forEach(el => {
        el.style.display = el.getAttribute('data-lang') === currentLang ? '' : 'none';
      });
    }

    // Interactive popover compiler (Fills session list dynamically) [1]
    window.openSyllabusModal = function(moduleName) {
      const activeSessions = sessionsCache.filter(s => s.module === moduleName);
      const body = document.getElementById('modLessonsModalBody');
      const modal = document.getElementById('modLessonsModal');

      if (!body || !modal) return;

      if (activeSessions.length === 0) {
        body.innerHTML = `
          <div style="text-align:center; padding: 20px; color: var(--text-light);">
            <i class="fa-regular fa-calendar-xmark" style="font-size:2rem; margin-bottom: 8px; display:block;"></i>
            <span data-lang="en">No classes scheduled on this module yet.</span>
            <span data-lang="fr">Aucune séance planifiée.</span>
            <span data-lang="ar">لا توجد أي حصص مبرمجة في هذا المقياس بعد.</span>
          </div>
        `;
      } else {
        body.innerHTML = activeSessions.map(s => `
          <div class="mod-modal-session-item">
            <div style="font-weight:800; color:var(--text); font-size:13.5px; margin-bottom: 4px;">${escapeHtml(s.module)}</div>
            <div style="font-size:11.5px; color:var(--text-secondary); display:flex; gap:10px; flex-wrap:wrap; font-weight:600; margin-top:2px;">
              <span><i class="fa-solid fa-users" style="color:var(--primary)"></i> ${escapeHtml(s.group)}</span>
              <span><i class="fa-solid fa-layer-group" style="color:var(--primary)"></i> ${s.type}</span>
              <span><i class="fa-solid fa-clock" style="color:var(--primary)"></i> ${s.time}</span>
              <span><i class="fa-solid fa-location-dot" style="color:var(--primary)"></i> ${escapeHtml(s.room)}</span>
            </div>
          </div>
        `).join('');
      }

      modal.classList.add('active');
      
      const currentLang = localStorage.getItem('acadtrack-lang') || 'en';
      modal.querySelectorAll('[data-lang]').forEach(el => {
        el.style.display = el.getAttribute('data-lang') === currentLang ? '' : 'none';
      });
    };

    window.closeSyllabusModal = function() {
      const modal = document.getElementById('modLessonsModal');
      if (modal) modal.classList.remove('active');
    };

    window.setModulesFilterQuery = function(val) {
      searchQuery = val;
      renderModulesGrid();
    };

    function escapeHtml(text) {
      if (text == null) return '';
      const div = document.createElement('div');
      div.textContent = String(text);
      return div.innerHTML;
    }

    // Bind custom panel listener transitions
    window.addEventListener('tabChanged', function(e) {
      if (e.detail && e.detail.panelId === 'panel-modules') {
        initModulesPanel();
      }
    });

    // Handle language switch updates
    window.addEventListener('load', () => {
      const observer = new MutationObserver((mutations) => {
        mutations.forEach((mutation) => {
          if (mutation.attributeName === "class" && document.getElementById('panel-modules')?.classList.contains('active')) {
            renderModulesGrid();
          }
        });
      });
      observer.observe(document.body, { attributes: true });
    });

    // Pre-load if active on first render
    if (document.getElementById('panel-modules')?.classList.contains('active')) {
      initModulesPanel();
    }
  })();
</script>

<script id="schedule-script">
  (function() {
    let studentObj = null;
    let sessionsCache = [];

    const weekdaysList = ["Sunday", "Monday", "Tuesday", "Wednesday", "Thursday"];

    const arabicDaysMap = {
      Sunday: "الأحد", Monday: "الإثنين", Tuesday: "الثلاثاء", Wednesday: "الأربعاء", Thursday: "الخميس"
    };

    const frenchDaysMap = {
      Sunday: "Dimanche", Monday: "Lundi", Tuesday: "Mardi", Wednesday: "Mercredi", Thursday: "Jeudi"
    };

    function initSchedulePanel() {
      loadScheduleData();
      renderDesktopGridTimetable();
      renderMobileTimelineList();
      updateWeeklyWorkloadStats();
    }

    function loadScheduleData() {
      const sessionUser = JSON.parse(localStorage.getItem('acadtrack_currentUser') || '{}');
      const students = window.getData('students') || [];
      
      studentObj = students.find(s => s.id === sessionUser.studentId) || students[0];
      
      const currentGroup = studentObj ? studentObj.group : 'L3 Psychologie Clinique';
      document.getElementById('sched-group-label').textContent = currentGroup;

      const sessions = window.getData('sessions') || [];
      sessionsCache = sessions.filter(s => s.group === currentGroup);
    }

    function updateWeeklyWorkloadStats() {
      const count = sessionsCache.length;
      document.getElementById('sched-stat-total').textContent = count;

      const hours = (count * 1.5).toFixed(1);
      document.getElementById('sched-stat-hours').textContent = hours;

      const uniqueModules = [...new Set(sessionsCache.map(s => s.module))].length;
      document.getElementById('sched-stat-modules').textContent = uniqueModules;

      const primaryRoom = sessionsCache.length > 0 ? sessionsCache[0].room : 'Salle 101';
      document.getElementById('sched-stat-room').textContent = primaryRoom;
    }

    function getSessionWeekdayName(dateString) {
      const days = ["Sunday", "Monday", "Tuesday", "Wednesday", "Thursday", "Friday", "Saturday"];
      const d = new Date(dateString);
      return days[d.getDay()];
    }

    function renderDesktopGridTimetable() {
      document.querySelectorAll('.timetable-cell.empty').forEach(cell => {
        cell.innerHTML = '';
        cell.style.background = '';
        cell.style.padding = '';
      });

      sessionsCache.forEach(sess => {
        const day = getSessionWeekdayName(sess.date);
        
        let slotTime = '08:30';
        if (sess.time >= '14:45') slotTime = '14:45';

    function translateSaaSElements(container) {

      const currentLang = localStorage.getItem('acadtrack-lang') || 'en';
      container.querySelectorAll('[data-lang]').forEach(el => {
        el.style.display = el.getAttribute('data-lang') === currentLang ? '' : 'none';
      });
    }

    function escapeHtml(text) {
      if (text == null) return '';
      const div = document.createElement('div');
      div.textContent = String(text);
      return div.innerHTML;
    }

        initSchedulePanel();

      initSchedulePanel();
    }

      const ex = parseFloat(exam || '0');
      const t_d = parseFloat(td || '0');
      const t_p = parseFloat(tp || '0');
      const ass = parseFloat(assignment || '0');
      const part = parseFloat(participation || '0');

      const average = (ex * 0.50) + (t_d * 0.20) + (t_p * 0.15) + (ass * 0.10) + (part * 0.05);
      return parseFloat(average.toFixed(2));
    }

    function calculateDistanceInMeters(lat1, lon1, lat2, lon2) {
      const R = 6371e3; // Earth radius in meters
      const phi1 = lat1 * Math.PI/180;
      const phi2 = lat2 * Math.PI/180;
      const deltaPhi = (lat2-lat1) * Math.PI/180;
      const deltaLon = (lon2-lon1) * Math.PI/180;

      const a = Math.sin(deltaPhi/2) * Math.sin(deltaPhi/2) +
                Math.cos(phi1) * Math.cos(phi2) *
                Math.sin(deltaLon/2) * Math.sin(deltaLon/2);
      const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a));
      return R * c;
    }

    };

    window.selectSaaSTargetContact = function(id) {
      activeContactId = id;
      
      // Mark as read
      messagesDatabase.forEach(m => {
        if (m.contactId === id && m.sender === 'received') m.unread = false;
      });

      renderContactsDirectory();

      const contactObj = contactsRoster.find(c => c.id === id);
      if (!contactObj) return;

      document.getElementById('activeChatName').textContent = contactObj.name;
      document.getElementById('activeChatStatus').style.display = 'inline';
      document.getElementById('chat-input-bar-form').style.display = 'flex';

      const portal = document.getElementById('msgPortalContainer');
      if (portal) portal.classList.add('chat-active');

      renderActiveConversationView();
    };

    window.backToInbox = function() {
      activeContactId = null;
      
      const portal = document.getElementById('msgPortalContainer');
      if (portal) portal.classList.remove('chat-active');
      
      renderContactsDirectory();
    };

    function renderActiveConversationView() {
      const body = document.getElementById('chatRosterBody');
      if (!body) return;

      const chatLogs = messagesDatabase.filter(m => m.contactId === activeContactId);

      if (chatLogs.length === 0) {
        body.innerHTML = `
          <div class="pt-center" style="height: 100%; display: flex; align-items: center; justify-content: center;">
            <div style="text-align: center; color: var(--text-light); font-size: 13px; font-weight: 600;">
              <span data-lang="en">No message logs. Start typing below...</span>
              <span data-lang="fr">Aucun message. Saisissez votre texte ci-dessous...</span>
              <span data-lang="ar">لا توجد سجلات رسائل بينكما. اكتب رسالة أدناه للبدء...</span>
            </div>
          </div>
        `;
        translateMessagesPanelElements();
        return;
      }

      body.innerHTML = chatLogs.map(m => {
        const wrapClass = m.sender === 'sent' ? 'sent' : 'received';
        const senderLabel = m.sender === 'sent' ? 'Me' : contactsRoster.find(c => c.id === m.contactId)?.name || 'Incoming';
        
        return `
          <div class="msg-bubble-wrap ${wrapClass}">
            <div class="msg-bubble">
              <span class="chat-sender">${escapeHtml(senderLabel)}</span>
              <span>${escapeHtml(m.text)}</span>
            </div>
            <span class="msg-timestamp">${m.date}</span>
          </div>
        `;
      }).join('');

      // Append hidden typing indicator bubble with Messenger CSS dots [2]
      body.innerHTML += `
        <div class="msg-bubble-wrap received" id="chat-typing-indicator" style="display:none; align-self: flex-start;">
          <div class="msg-bubble" style="background:#ffffff; border:1px solid #e2e8f0; padding: 12px 18px; display:flex; align-items:center; gap:4px; border-bottom-left-radius: 2px;">
            <span style="font-size:11px; font-weight:700; color:#94a3b8;" id="typing-text">تكتب الآن</span>
            <div class="typing-dots">
              <span class="typing-dot"></span>
              <span class="typing-dot"></span>
              <span class="typing-dot"></span>
            </div>
          </div>
        </div>
      `;

      body.scrollTop = body.scrollHeight; // Force Scroll to Bottom
    }

    // Submit message and trigger live typing simulation loop [2]
    window.submitLiveSaaSMessage = function() {
      const input = document.getElementById('chatViewportInputField');
      if (!input || !activeContactId) return;

      const text = input.value.trim();
      if (!text) return;

      const timeStr = new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });

      const newMsg = {
        contactId: activeContactId,
        sender: 'sent',
        text: text,
        date: timeStr,
        unread: false
      };

      messagesDatabase.push(newMsg);

      input.value = ''; // Reset
      renderActiveConversationView();
      renderContactsDirectory();

      // Trigger automatic sandbox simulated replying delay with typing animation [2]
      simulateAdminBypassReply(activeContactId);
    };

    function simulateAdminBypassReply(contactId) {
      const targetContact = contactsRoster.find(c => c.id === contactId);
      if (!targetContact) return;

      // SaaS Darija response structures addressing as "يا عماد" (NO emojis used) [2]
      const responsePool = {
        Admin: [
          "صحيت يا عماد راني شفت الملفات تاعك وكلش راهو مريغل في البوابة الأكاديمية",

          "يا عماد ما تنساش برك تبعثلي الملف النهائي قبل نهاية الأسبوع عيشك"
        ],
        Teacher: [

          "يعيشك يا عماد غدوة مع العشرة راني في البيرو تاعي في القسم نتلاقاو تما ونشوفوه مع بعضنا",
          "الله يحفظك يا عماد راني شفت الرسالة تاعك كلش راهو واجد ذرك يعطيك الصحة"
        ],
        Student: [
          "صحيت يا عماد يرحم والديك ذرك نبعثهولك في البوابة الأكاديمية حالا",
          "يعيشك يا عماد راني مع صحابنا نوجدوا في العرض وبإذن الله نبعثوه في الوقت المحدد",
          "صحيت يا عماد يكثر خيرك رانا نتبعو في الحصص ونعوضو في الغيابات اللي فاتونا"
        ]
      };

      const pool = responsePool[targetContact.type] || responsePool['Admin'];
      const chosenReply = pool[Math.floor(Math.random() * pool.length)];

      const body = document.getElementById('chatRosterBody');
      const indicator = document.getElementById('chat-typing-indicator');
      const typingText = document.getElementById('typing-text');

      // 1. Show Bouncing Dots Typing Animation [2]
      if (indicator && typingText) {
        typingText.textContent = `${targetContact.name} يكتب الآن`;
        indicator.style.display = 'flex';
        if (body) body.scrollTop = body.scrollHeight;
      }

      // 2. Ticks delay 1.5s before injecting message [2]
      setTimeout(() => {
        if (indicator) indicator.style.display = 'none';

        const replyTime = new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });
        
        const replyMsg = {
          contactId: contactId,
          sender: 'received',
          text: chosenReply,
          date: replyTime,
          unread: true
        };

        messagesDatabase.push(replyMsg);

        if (activeContactId === contactId) {
          renderActiveConversationView();
          replyMsg.unread = false;
        }

        renderContactsDirectory();

      }, 1500);
    }

    function translateMessagesPanelElements() {
      const currentLang = localStorage.getItem('acadtrack-lang') || 'en';
      document.querySelectorAll('#panel-messages [data-lang]').forEach(el => {
        el.style.display = el.getAttribute('data-lang') === currentLang ? '' : 'none';
      });
    }

    function escapeHtml(text) {
      if (text == null) return '';
      const div = document.createElement('div');
      div.textContent = String(text);
      return div.innerHTML;
    }

    // Bind custom panel listener transitions
    window.addEventListener('tabChanged', function(e) {
      if (e.detail && e.detail.panelId === 'panel-messages') {
        initMessagesPanel();
      }
    });

    // Handle language switches dynamically
    window.addEventListener('load', () => {
      const observer = new MutationObserver((mutations) => {
        mutations.forEach((mutation) => {
          if (mutation.attributeName === "class" && document.getElementById('panel-messages')?.classList.contains('active')) {
            loadMessagesDatabase();
            renderContactsDirectory();
            if (activeContactId) {
              selectSaaSTargetContact(activeContactId);
            }
          }
        });
      });
      observer.observe(document.body, { attributes: true });
    });

    // Pre-load if active on first render
    if (document.getElementById('panel-messages')?.classList.contains('active')) {
      initMessagesPanel();
    }
  })();
</script>

<script id="calendar-script">
  // ==========================================
  // MODULE: ACADEMIC CALENDAR CONTROLLER
  // ==========================================
  (function() {
    let studentObj = null;
    let sessionsCache = [];

    // State tracks managed navigated months (Defaulting to May 2026 context)
    let calendarState = {
      month: 4, // May
      year: 2026
    };

    function initCalendarPanel() {
      loadCalendarData();
      renderMonthlyCalendarGrid();
    }

    function loadCalendarData() {
      const sessionUser = JSON.parse(localStorage.getItem('acadtrack_currentUser') || '{}');
      const students = window.getData('students') || [];
      
      studentObj = students.find(s => s.id === sessionUser.studentId) || students[0];

      const sessions = window.getData('sessions') || [];
      const currentGroup = studentObj ? studentObj.group : 'L3 Psychologie Clinique';
      
      // Filter sessions matching specifically the student's cohort group [1]
      sessionsCache = sessions.filter(s => s.group === currentGroup);
    }

    function renderMonthlyCalendarGrid() {
      const year = calendarState.year;
      const month = calendarState.month;

      // Sync Month Title Label
      const titleLabel = document.getElementById('cal-month-year-title');
      if (titleLabel) {
        const tempDate = new Date(year, month, 1);
        titleLabel.innerHTML = `
          <span data-lang="en">${tempDate.toLocaleDateString('en-US', { month: 'long', year: 'numeric' })}</span>
          <span data-lang="fr">${tempDate.toLocaleDateString('fr-FR', { month: 'long', year: 'numeric' })}</span>
          <span data-lang="ar">${tempDate.toLocaleDateString('ar-SA', { month: 'long', year: 'numeric' })}</span>
        `;
      }

      const gridNode = document.getElementById('calDaysInsertionNode');
      if (!gridNode) return;

      const firstDayIndex = new Date(year, month, 1).getDay();
      const daysInMonth = new Date(year, month + 1, 0).getDate();
      const prevMonthDays = new Date(year, month, 0).getDate();

      const today = new Date();
      today.setHours(0,0,0,0);

      let gridHTML = '';

      // trailing days
      for (let i = firstDayIndex - 1; i >= 0; i--) {
        gridHTML += `<div class="cal-canvas-day-cell other-month"><span class="cal-cell-day-num">${prevMonthDays - i}</span></div>`;
      }

      // days in month
      for (let day = 1; day <= daysInMonth; day++) {
        const dateString = `${year}-${String(month + 1).padStart(2, '0')}-${String(day).padStart(2, '0')}`;
        
        // Filter session matches
        const activeSessions = sessionsCache.filter(s => s.date === dateString);

        const targetDate = new Date(year, month, day);
        targetDate.setHours(0,0,0,0);
        const isToday = targetDate.getTime() === today.getTime();

        const todayClass = isToday ? 'today' : '';

        const eventChipsHTML = activeSessions.map(s => `
          <div class="cal-event-chip ${s.type.toLowerCase()}">${escapeHtml(s.module)} (${s.type})</div>
        `).join('');

        gridHTML += `
          <div class="cal-canvas-day-cell ${todayClass}" onclick="openCalDetailsModal('${dateString}')">
            <span class="cal-cell-day-num">${day}</span>
            <div class="cal-cell-events-list">
              ${eventChipsHTML}
            </div>
          </div>
        `;
      }

      // leading days
      const totalCells = firstDayIndex + daysInMonth;
      const remainingCells = (7 - (totalCells % 7)) % 7;
      for (let i = 1; i <= remainingCells; i++) {
        gridHTML += `<div class="cal-canvas-day-cell other-month"><span class="cal-cell-day-num">${i}</span></div>`;
      }

      gridNode.innerHTML = gridHTML;
      translateSaaSElements(gridNode);
    }

    window.navigateActiveCalendarMonth = function(delta) {
      calendarState.month += delta;
      if (calendarState.month > 11) {
        calendarState.month = 0;
        calendarState.year++;
      }
      if (calendarState.month < 0) {
        calendarState.month = 11;
        calendarState.year--;
      }
      renderMonthlyCalendarGrid();
    };

  (function() {

    function loadNotificationPreferences() {
      const emailOpt = localStorage.getItem('saas_pref_email') !== 'false';
      const smsOpt   = localStorage.getItem('saas_pref_sms') === 'true'; 
      const pushOpt  = localStorage.getItem('saas_pref_push') !== 'false';

      document.getElementById('pref-opt-email').checked = emailOpt;
      document.getElementById('pref-opt-sms').checked = smsOpt;
      document.getElementById('pref-opt-push').checked = pushOpt;
    }

    window.saveNotificationPreferencesState = function() {
      const email = document.getElementById('pref-opt-email').checked;
      const sms   = document.getElementById('pref-opt-sms').checked;
      const push  = document.getElementById('pref-opt-push').checked;

      localStorage.setItem('saas_pref_email', email);
      localStorage.setItem('saas_pref_sms', sms);
      localStorage.setItem('saas_pref_push', push);

      triggerSuccessToast("Preferences Saved", "Your workspace notification options have been updated.");
    };

    // ==========================================
    // BACKUP MANAGEMENT (EXPORT, IMPORT, RESET)
    // ==========================================
    window.exportSaaSDatabase = function() {
      const dataBackupObject = {
        students: localStorage.getItem('students'),
        modules: localStorage.getItem('modules'),
        groups: localStorage.getItem('groups'),
        sessions: localStorage.getItem('sessions'),
        grades: localStorage.getItem('grades'),
        reports: localStorage.getItem('reports'),
        attendance: localStorage.getItem('attendance'),
        messages: localStorage.getItem('messages')
      };

      const fileBlob = new Blob([JSON.stringify(dataBackupObject, null, 2)], { type: 'application/json' });
      const link = document.createElement('a');
      link.href = URL.createObjectURL(fileBlob);
      link.download = `acadtrack_student_database_backup_${Date.now()}.json`;
      link.click();
    };

    window.restoreSaaSDatabaseFromJSON = function(event) {
      const file = event.target.files[0];
      if (!file) return;

      const reader = new FileReader();
      reader.onload = function(e) {
        try {
          const parsed = JSON.parse(e.target.result);
          
          const dbKeys = ['students', 'modules', 'groups', 'sessions', 'grades', 'reports', 'attendance'];
          const isValidBackup = dbKeys.some(key => parsed.hasOwnProperty(key));

          if (!isValidBackup) {
            alert("Restoration Failed: Selected file is not a valid AcadTrack backup.");
            return;
          }

          dbKeys.forEach(key => {
            if (parsed[key] !== undefined && parsed[key] !== null) {
              localStorage.setItem(key, parsed[key]);
            }
          });
          
          if (parsed.messages) localStorage.setItem('messages', parsed.messages);

          triggerSuccessToast("Backup Restored", "Personal database successfully uploaded.");
          
        } catch(err) {
          alert("Error: Failed to parse database file structures.");
        }
      };
      reader.readAsText(file);
    };

    window.clearAndReSeedDatabase = function() {
      const confirmText = "Are you sure you want to completely wipe all current department rosters, groups, and teachers? This will restore Batna 1 default databases.";
      if (confirm(confirmText)) {
        localStorage.clear();
        
        // Re-run global seed database
        seedDatabase();
        loadNotificationPreferences();
        
        triggerSuccessToast("Database Wiped", "Department baseline parameters re-seeded successfully.");
      }
    };

    // Spans floating green success toast alerts (Top-Left) [1]
    function triggerSuccessToast(title, description) {
      const container = document.getElementById('profileToastContainer');
      if (!container) return;

      const currentLang = localStorage.getItem('acadtrack-lang') || 'en';
      const toastId = 'toast_' + Date.now();

      const toastHTML = `
        <div class="saas-toast-alert" id="${toastId}">
          <div class="saas-toast-icon">
            <i class="fa-solid fa-circle-check"></i>
          </div>
          <div class="saas-toast-body">
            <div class="saas-toast-title">${title}</div>
            <div class="saas-toast-desc">${description}</div>
          </div>
        </div>
      `;

      container.innerHTML += toastHTML;

      // Slide-in transition [1]
      setTimeout(() => {
        const el = document.getElementById(toastId);
        if (el) el.classList.add('active');
      }, 50);

      // Auto Dismiss after 4 seconds
      setTimeout(() => {
        const el = document.getElementById(toastId);
        if (el) {
          el.classList.remove('active');
          setTimeout(() => el.remove(), 350);
        }
      }, 4000);
    }

    function translateSaaSElements(container) {
      if (!container) return;
      const currentLang = localStorage.getItem('acadtrack-lang') || 'en';
      container.querySelectorAll('[data-lang]').forEach(el => {
        el.style.display = el.getAttribute('data-lang') === currentLang ? '' : 'none';
      });
    }

    // Bind custom panel listener transitions
    window.addEventListener('tabChanged', function(e) {
      if (e.detail && e.detail.panelId === 'panel-settings') {
        initSettingsPanel();
      }
    });

    // Handle language switch updates
    window.addEventListener('load', () => {
      const observer = new MutationObserver((mutations) => {
        mutations.forEach((mutation) => {
          if (mutation.attributeName === "class" && document.getElementById('panel-settings')?.classList.contains('active')) {
            initSettingsPanel();
          }
        });
      });
      observer.observe(document.body, { attributes: true });
    });

    // Pre-load if active on first render
    if (document.getElementById('panel-settings')?.classList.contains('active')) {
      initSettingsPanel();
    }
  })();
