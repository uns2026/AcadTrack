      alert(msgErr[lang] || msgErr.en);
      return;
    }

    if (!newPass || newPass.length < 6) {

        en: "Security Error: New password must be at least 6 characters.",
        fr: "Erreur Sécurité : Le mot de passe doit comporter au moins 6 caractères.",

      return;
    }

    if (newPass !== confirmPass) {

        en: "Security Error: Confirm password does not match new password.",
        fr: "Erreur Sécurité : La confirmation ne correspond pas au nouveau mot de passe.",
        ar: "خطأ أمني: تأكيد كلمة المرور غير متطابق مع كلمة المرور الجديدة."
