  let currentStep = 0;
  let selectedRole = '';
  
  const validationErrors = {
    role_required: { en: "Please select an account type first.", fr: "Veuillez d'abord sélectionner un type de compte.", ar: "يرجى تحديد نوع الحساب أولاً." },
    fields_required: { en: "Please fill out all required fields.", fr: "Veuillez remplir tous les champs obligatoires.", ar: "يرجى ملء جميع الحقول المطلوبة." },
    password_length: { en: "Password must be at least 6 characters.", fr: "Le mot de passe doit comporter au moins 6 caractères.", ar: "يجب أن تتكون كلمة المرور من 6 أحرف على الأقل." },
    password_mismatch: { en: "Passwords do not match.", fr: "Les mots de passe ne correspondent pas.", ar: "كلمات المرور غير متطابقة." },
    invalid_email: { en: "Invalid email address format.", fr: "Format d'adresse e-mail invalide.", ar: "صيغة البريد الإلكتروني غير صالحة." },
    email_exists: { en: "This email address is already registered. Please use a different email.", fr: "Cette adresse e-mail est déjà enregistrée. Veuillez utiliser un autre e-mail.", ar: "هذا البريد الإلكتروني مسجل بالفعل. يرجى استخدام بريد إلكتروني مختلف." },
    phone_exists: { en: "This phone number is already registered. Please use a different number.", fr: "Ce numéro de téléphone est déjà enregistré. Veuillez utiliser un autre numéro.", ar: "رقم الهاتف هذا مسجل بالفعل. يرجى استخدام رقم هاتف مختلف." },
    phone_numeric: { en: "Only numbers are accepted in the phone field.", fr: "Seuls les chiffres sont acceptés dans le champ téléphonique.", ar: "يُقبل فقط الأرقام في حقل الهاتف." },
    phone_length: { en: "Phone number must be exactly 10 digits.", fr: "Le numéro de téléphone doit comporter exactement 10 chiffres.", ar: "يجب أن يتكون رقم الهاتف من 10 أرقام بالضبط." }
  };

  // Initialize App
  document.addEventListener('DOMContentLoaded', function() {
    const cachedRole = localStorage.getItem('acadtrack_signup_role');
    if (cachedRole) selectRole(cachedRole);

    const initLang = localStorage.getItem('acadtrack-lang') || 'en';
    setLanguage(initLang);

    // Event Delegation
    document.querySelector('.onboarding-card').addEventListener('click', function(e) {
      const target = e.target.closest('button, .role-select-card');
      if (!target) return;

      switch(target.id) {
        case 'btnWelcomeStart': navigateOnboarding(1); break;
        case 'cardTeacher': selectRole('teacher'); break;
        case 'cardAdmin': selectRole('admin'); break;
        case 'btnRoleBack': navigateOnboarding(0); break;
        case 'btnRoleContinue': navigateOnboarding(2); break;
        case 'btnFormBack': navigateOnboarding(1); break;
        case 'btnGoogleSignup': simulateGoogleAuthLoop(); break;
        case 'btnConfirmBack': navigateOnboarding(2); break;
        case 'btnConfirmSubmit': handleFinalSubmit(); break;
      }
    });

    document.getElementById('onboardingSignupForm').addEventListener('submit', function(e) {
      e.preventDefault();
      validateAndPreview();
    });

    // Real-time Input Formatting
    const phoneInput = document.getElementById('userPhone');
    const firstNameInput = document.getElementById('userFirstName');
    const lastNameInput = document.getElementById('userLastName');

    phoneInput.addEventListener('input', function(e) {
      e.target.value = e.target.value.replace(/[^0-9]/g, '');
    });

    firstNameInput.addEventListener('input', function(e) {
      let val = e.target.value;
      if(val.length > 0) {
        e.target.value = val.charAt(0).toUpperCase() + val.slice(1).toLowerCase();
      }
    });

    lastNameInput.addEventListener('input', function(e) {
      e.target.value = e.target.value.toUpperCase();
    });

    navigateOnboarding(0);
  });

  // Multilingual Engine - Dynamic RTL/LTR Logic
  function setLanguage(lang) {
    localStorage.setItem('acadtrack-lang', lang);
    document.documentElement.lang = lang;
    
    // Force RTL strictly on the onboarding card when Arabic is selected
    const isRTL = lang === 'ar';
    const onboardingCard = document.getElementById('mainOnboardingCard');
    if (onboardingCard) {
        onboardingCard.dir = isRTL ? 'rtl' : 'ltr';
    }
    
    document.querySelectorAll('[data-lang]').forEach(el => {
      el.style.display = el.dataset.lang === lang ? '' : 'none';
    });
  }

  // Step Navigation
  function navigateOnboarding(stepNum) {
    currentStep = stepNum;
    document.getElementById('errorBox').style.display = 'none';

    document.querySelectorAll('.onboarding-step').forEach(step => step.classList.remove('active'));
    const activeStep = document.getElementById('step' + stepNum);
    if (activeStep) activeStep.classList.add('active');

    const progressContainer = document.getElementById('stepProgressContainer');
    const progressLabel = document.getElementById('stepProgressLabel');
    
    if (stepNum === 0) {
      progressContainer.style.display = 'none';
    } else {
      progressContainer.style.display = 'block';
      const currentLang = localStorage.getItem('acadtrack-lang') || 'en';
      const labels = { en: `Step ${stepNum} of 3`, fr: `Étape ${stepNum} sur 3`, ar: `الخطوة ${stepNum} من 3` };
      progressLabel.textContent = labels[currentLang];
    }

    document.querySelectorAll('.step-dot').forEach((dot, index) => {
      dot.classList.toggle('active', index + 1 <= stepNum);
    });
    document.querySelectorAll('.step-line').forEach((line, index) => {
      line.classList.toggle('active', index + 2 <= stepNum);
    });
  }

  // Role Selection
  function selectRole(role) {
    selectedRole = role;
    localStorage.setItem('acadtrack_signup_role', role);

    document.querySelectorAll('.role-select-card').forEach(c => c.classList.remove('selected'));
    const selectedCard = document.getElementById(role === 'teacher' ? 'cardTeacher' : 'cardAdmin');
    if (selectedCard) selectedCard.classList.add('selected');

    const btnContinue = document.getElementById('btnRoleContinue');
    btnContinue.disabled = false;
    btnContinue.style.cursor = 'pointer';
    btnContinue.style.opacity = '1';
  }

  // Password Toggle
  function togglePasswordVisibility(fieldId) {
    const field = document.getElementById(fieldId);
    const eyeIcon = document.getElementById(fieldId + 'Eye');
    if (field.type === 'password') {
      field.type = 'text';
      eyeIcon.className = 'fa-solid fa-eye-slash';
    } else {
      field.type = 'password';
      eyeIcon.className = 'fa-solid fa-eye';
    }
  }

  // Simulated Google OAuth
  function simulateGoogleAuthLoop() {
    const overlay = document.getElementById('loadingOverlay');
    overlay.querySelector('.onboarding-spinner').style.display = 'block';
    overlay.querySelector('.loader-text').innerHTML = '<span data-lang="en">Connecting to Google...</span><span data-lang="fr">Connexion à Google...</span><span data-lang="ar">جاري الاتصال بـ Google...</span>';
    overlay.style.display = 'flex';
    setLanguage(localStorage.getItem('acadtrack-lang') || 'en');

    setTimeout(function() {
      overlay.style.display = 'none';

      document.getElementById('userTitle').value = 'Pr.';
      document.getElementById('userFirstName').value = 'Zakia';
      document.getElementById('userLastName').value = 'CHENNA';
      document.getElementById('userEmail').value = 'zakia.chenna@univ-batna.dz';
      document.getElementById('userPhone').value = '0659144039';
      document.getElementById('userPassword').value = 'uns2026';
      document.getElementById('userConfirmPassword').value = 'uns2026';
      document.getElementById('userInstitution').value = 'Batna 1 University';
      document.getElementById('userDepartment').value = 'Psychology department';

      document.getElementById('googleBtnContainer').style.display = 'none';
      document.getElementById('googleAuthSuccessBox').style.display = 'flex';

      navigateOnboarding(2);
    }, 1200);
  }

  // Error Handling
  function dispatchFormError(errorKey) {
    const errorBox = document.getElementById('errorBox');
    const errorMsgText = document.getElementById('errorMsgText');
    const currentLang = localStorage.getItem('acadtrack-lang') || 'en';
    
    errorMsgText.textContent = validationErrors[errorKey][currentLang] || validationErrors[errorKey]['en'];
    errorBox.style.display = 'flex';
    window.scrollTo({ top: 0, behavior: 'smooth' });
  }

  // Validate Step 2 and Populate Step 3
  function validateAndPreview() {
    const password = document.getElementById('userPassword').value;
    const confirmPassword = document.getElementById('userConfirmPassword').value;
    const email = document.getElementById('userEmail').value.trim();
    const phone = document.getElementById('userPhone').value.trim();

    if (!selectedRole) return dispatchFormError('role_required');
    
    let allFilled = true;
    document.querySelectorAll('#onboardingSignupForm input, #onboardingSignupForm select').forEach(inp => {
      if(inp.required && !inp.value.trim()) allFilled = false;
    });
    if (!allFilled) return dispatchFormError('fields_required');

    if (!/^\d+$/.test(phone)) return dispatchFormError('phone_numeric');
    if (phone.length !== 10) return dispatchFormError('phone_length');

    if (password.length < 6) return dispatchFormError('password_length');
    if (password !== confirmPassword) return dispatchFormError('password_mismatch');
    
    if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) return dispatchFormError('invalid_email');

    // --- DUPLICATE CHECK BEFORE MOVING TO STEP 3 ---
    let localUsers = [];
    try {
      const stored = localStorage.getItem('acadtrack_users');
      if (stored) localUsers = JSON.parse(stored);
    } catch (e) { localUsers = []; }

    const emailExists = localUsers.some(u => u.email.toLowerCase() === email.toLowerCase());
    if (emailExists) return dispatchFormError('email_exists'); // Blocks step transition

    const phoneExists = localUsers.some(u => u.phone === phone);
    if (phoneExists) return dispatchFormError('phone_exists'); // Blocks step transition
    // -----------------------------------------------

    const title = document.getElementById('userTitle').value;
    const firstName = document.getElementById('userFirstName').value.trim();
    const lastName = document.getElementById('userLastName').value.trim();

    document.getElementById('confirmUserName').textContent = `${title} ${firstName} ${lastName}`;
    document.getElementById('confirmRole').textContent = selectedRole === 'teacher' ? 'Teacher Account' : 'Department Admin';
    document.getElementById('confirmEmail').textContent = email;
    document.getElementById('confirmPhone').textContent = phone;
    document.getElementById('confirmInstitution').textContent = document.getElementById('userInstitution').value.trim();
    document.getElementById('confirmDepartment').textContent = document.getElementById('userDepartment').value.trim();

    navigateOnboarding(3);
  }

  // Final Submit Logic
  function handleFinalSubmit() {
    // Show Loader and Countdown immediately
    const overlay = document.getElementById('loadingOverlay');
    overlay.querySelector('.onboarding-spinner').style.display = 'block';
    overlay.querySelector('.loader-text').innerHTML = 'Redirecting in <span id="countdownTimer">5</span>s...';
    overlay.style.display = 'flex';

    let timeLeft = 5;
    const timerElement = overlay.querySelector('#countdownTimer');
    
    const countdownInterval = setInterval(() => {
      timeLeft--;
      timerElement.textContent = timeLeft;
      
      if (timeLeft <= 0) {
        clearInterval(countdownInterval);
        saveDataToLocalStorage();
        window.location.href = '/p/subscribe.html';
      }
    }, 1000);
  }

  // Local Storage Write Operation
  function saveDataToLocalStorage() {
    const title = document.getElementById('userTitle').value;
    const firstName = document.getElementById('userFirstName').value.trim();
    const lastName = document.getElementById('userLastName').value.trim();
    const email = document.getElementById('userEmail').value.trim();
    const phone = document.getElementById('userPhone').value.trim();
    const password = document.getElementById('userPassword').value;
    const institution = document.getElementById('userInstitution').value.trim();
    const department = document.getElementById('userDepartment').value.trim();

    let localUsers = [];
    try {
      const stored = localStorage.getItem('acadtrack_users');
      if (stored) localUsers = JSON.parse(stored);
    } catch (e) { localUsers = []; }

    const mockHashedPassword = btoa(password); 

    const now = new Date();
    const newUser = {
      uid: 'user_' + now.getTime(),
      title, firstName, lastName,
      name: `${title} ${firstName} ${lastName}`,
      email, phone,
      password: mockHashedPassword, 
      institution, department,
      role: selectedRole,
      status: "pending_subscription",
      approved: false,
      paymentStatus: "unpaid",
      createdAt: now.toISOString(),
      createdAtFormatted: now.toLocaleString()
    };

    localUsers.push(newUser);
    localStorage.setItem('acadtrack_users', JSON.stringify(localUsers));
    localStorage.setItem('acadtrack_currentUser', JSON.stringify(newUser));
    localStorage.setItem('acadtrack_demoMode', 'true');
    localStorage.removeItem('acadtrack_signup_role');
  }
