  function displayAuthError(key) {

  // Robust Decoder helper for Base64 password matches (Ensures compatibility with btoa/atob signup parameters)
  function isPasswordCorrect(enteredPass, storedPass) {
    if (enteredPass === storedPass) return true;
    try {
      if (btoa(enteredPass) === storedPass) return true;
    } catch(e) {}
    try {
      if (enteredPass === atob(storedPass)) return true;
    } catch(e) {}
    return false;
  }

  function handleTeacherSubmit(e) {
    e.preventDefault();
    document.getElementById('errorBox').style.display = 'none';

    const loginInput = document.getElementById('tEmail').value.trim().toLowerCase();
    const password = document.getElementById('tPassword').value;

    if (!loginInput || !password) {
      displayAuthError('empty_fields');
      return;
    }

    const overlay = document.getElementById('loadingOverlay');
    overlay.style.display = 'flex';

    setTimeout(function() {
      overlay.style.display = 'none';

      // Lookup inside Local Users database
      let usersList = [];
      try {
        const stored = localStorage.getItem('acadtrack_users');
        if (stored) usersList = JSON.parse(stored);
      } catch (err) { usersList = []; }

      const user = usersList.find(u => 
        (u.email.toLowerCase() === loginInput || u.phone === loginInput) && isPasswordCorrect(password, u.password)
      );

      if (!user) {
        displayAuthError('invalid_credentials');
        return;
      }

      // Store Active Session cache
      localStorage.setItem('acadtrack_currentUser', JSON.stringify(user));
      localStorage.setItem('acadtrack_demoMode', 'true');

      // Check workspace approval status parameters
      if (user.status === 'pending_subscription') {
        window.location.href = '/p/subscribe.html';
      } else {
        // Active Dashboard routing
        const targetDashboard = user.role === 'admin' ? '/p/dept-admin-dashboard.html' : '/p/teacher-dashboard.html';
        window.location.href = targetDashboard;
      }

    }, 1200);
  }

  function handleStudentSubmit(e) {
    e.preventDefault();
    document.getElementById('errorBox').style.display = 'none';

    const loginInput = document.getElementById('sId').value.trim();
    const password = document.getElementById('sPassword').value;

    if (!loginInput || !password) {
      displayAuthError('empty_fields');
      return;
    }

    const overlay = document.getElementById('loadingOverlay');
    overlay.style.display = 'flex';

    setTimeout(function() {
      overlay.style.display = 'none';

      let studentsList = [];
      try {
        const stored = localStorage.getItem('students');
        if (stored) studentsList = JSON.parse(stored);
      } catch (err) { studentsList = []; }

      // Lookup student record by ID or Phone or Email
      const student = studentsList.find(s => 
        s.id === loginInput || s.email === loginInput || s.phone === loginInput
      );

      if (!student) {
        displayAuthError('student_not_found');
        return;
      }

      // Check first-time login substate
      if (!student.completed) {
        // Must use correct default password to enter completion wizard
        const defaultPassword = student.defaultPassword || '123';
        if (password !== defaultPassword) {
          displayAuthError('student_password_incorrect');
          return;
        }

        // Trigger step-up Account Onboarding completing Wizard
        completingStudentObj = student;
        displayAccountCompletionWizard(student);
      } else {
        // Subsequent login - verify actual password
        if (!isPasswordCorrect(password, student.password)) {
          displayAuthError('invalid_credentials');
          return;
        }

        // Store active Student session
        const studentUserNode = {
          uid: 'student_' + student.id,
          name: student.firstName + ' ' + student.lastName,
          role: 'student',
          studentId: student.id,
          department: 'Psychologie',
          group: student.group,
          status: 'active'
        };
        localStorage.setItem('acadtrack_currentUser', JSON.stringify(studentUserNode));
        localStorage.setItem('acadtrack_demoMode', 'true');

        window.location.href = '/p/student-dashboard.html';
      }

    }, 1200);
  }

  // Step-up wizard trigger
  function displayAccountCompletionWizard(student) {
    document.getElementById('authStateView').style.display = 'none';
    const card = document.getElementById('authContainer');
    if (card) card.classList.add('completion-card');

    document.getElementById('cId').value = student.id;
    document.getElementById('cFirst').value = student.firstName;
    document.getElementById('cLast').value = student.lastName;
    document.getElementById('cGroup').value = student.group || 'L3 Clinique A';

    document.getElementById('accountCompletionView').style.display = 'block';
  }

  function handleCompletionSubmit(e) {
    e.preventDefault();
    document.getElementById('errorBox').style.display = 'none';

    const gender = document.getElementById('cGender').value;
    const dob = document.getElementById('cDob').value;
    const email = document.getElementById('cEmail').value.trim();
    const phone = document.getElementById('cPhone').value.trim();
    const password = document.getElementById('cPassword').value;

    if (!email || !phone || !password || !dob) {
      displayAuthError('empty_fields');
      return;
    }
    if (password.length < 6) {
      displayAuthError('password_length');
      return;
    }
    if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
      displayAuthError('invalid_email');
      return;
    }

    const overlay = document.getElementById('loadingOverlay');
    overlay.querySelector('.loader-text').innerHTML = '<span data-lang="en">Configuring profile parameters...</span><span data-lang="fr">Configuration de votre espace...</span><span data-lang="ar">جاري إعداد وتأمين الحساب...</span>';
    setLanguage(localStorage.getItem('acadtrack-lang') || 'en');
    overlay.style.display = 'flex';

    setTimeout(function() {
      overlay.style.display = 'none';

      // Load existing students DB, update profile nodes
      let studentsList = [];
      try {
        const stored = localStorage.getItem('students');
        if (stored) studentsList = JSON.parse(stored);

        studentsList[idx].email = email;
        studentsList[idx].phone = phone;

        localStorage.setItem('students', JSON.stringify(studentsList));
      }

      // Store Active Student Session Cache
      const studentUserNode = {
        uid: 'student_' + completingStudentObj.id,
        name: completingStudentObj.firstName + ' ' + completingStudentObj.lastName,
        role: 'student',
        studentId: completingStudentObj.id,
        department: 'Psychologie',
        group: completingStudentObj.group || 'L3 Clinique A',
        status: 'active'
      };
      localStorage.setItem('acadtrack_currentUser', JSON.stringify(studentUserNode));
      localStorage.setItem('acadtrack_demoMode', 'true');

      // Success Redirect
      window.location.href = '/p/student-dashboard.html';

    }, 1500);
