      cb.checked = !cb.checked;
      cardEl.classList.toggle('selected', cb.checked);
      calculateLivePricing();
    }
  }

  // Live dynamic pricing calculator (SaaS math based on Algeria specifications)
  function calculateLivePricing() {
    const sessionUser = JSON.parse(localStorage.getItem('acadtrack_currentUser') || '{}');
    const role = sessionUser.role || 'teacher';

    let basePrice = 0;
    let usageCost = 0;
    let featuresCost = 0;

    if (role === 'teacher') {
      basePrice = 1500;
      
      const modules = parseInt(document.getElementById('teacherModules')?.value || '1');
      usageCost += (modules * 200);

      const students = document.getElementById('teacherStudents')?.value || '1-50';
      if (students === '51-150') usageCost += 1000;
      else if (students === '151-300') usageCost += 2000;
      else if (students === '300+') usageCost += 4000;

      document.querySelectorAll('.teacher-feature-checkbox').forEach(cb => {
        if (cb.checked) featuresCost += parseInt(cb.dataset.cost || '0');
      });
    } else {
      basePrice = 5000;

      const teachers = parseInt(document.getElementById('adminTeachers')?.value || '1');
      const students = parseInt(document.getElementById('adminStudents')?.value || '1');

      usageCost += (teachers * 500);
      usageCost += (students * 15);

      document.querySelectorAll('.admin-feature-checkbox').forEach(cb => {
        if (cb.checked) featuresCost += parseInt(cb.dataset.cost || '0');
      });
    }

    priceObject.monthly = basePrice + usageCost + featuresCost;
    priceObject.semester = Math.round(priceObject.monthly * 6 * 0.90); // Save 10%
    priceObject.yearly = Math.round(priceObject.monthly * 12 * 0.80); // Save 20%

    updatePricingPanelUI();
  }

  // Sync pricing panel views
  function updatePricingPanelUI() {
    const displayEl = document.getElementById('pricingDisplay');
    const cycleEl = document.getElementById('pricingBilledCycle');
    const currentLang = localStorage.getItem('acadtrack-lang') || 'en';

    if (billingCycle === 'monthly') {
      selectedCyclePriceTotal = priceObject.monthly;
      displayEl.innerHTML = `${priceObject.monthly.toLocaleString()} DZD<span>/mo</span>`;
      cycleEl.textContent = currentLang === 'ar' ? 'فوترة شهرية' : currentLang === 'fr' ? 'Facturé mensuellement' : 'Billed monthly';
    } else if (billingCycle === 'semester') {
      selectedCyclePriceTotal = priceObject.semester;
      displayEl.innerHTML = `${priceObject.semester.toLocaleString()} DZD<span>/sem</span>`;
      cycleEl.textContent = currentLang === 'ar' ? 'وفر 10% - فوترة نصف سنوية' : currentLang === 'fr' ? 'Économisez 10% - Facturé par semestre' : 'Save 10% - Billed semi-annually';
    } else {
      selectedCyclePriceTotal = priceObject.yearly;
      displayEl.innerHTML = `${priceObject.yearly.toLocaleString()} DZD<span>/yr</span>`;
      cycleEl.textContent = currentLang === 'ar' ? 'وفر 20% - فوترة سنوية' : currentLang === 'fr' ? 'Économisez 20% - Facturé annuellement' : 'Save 20% - Billed annually';
    }
  }

  // Handle 3-way billing cycle switch buttons
  function switchBillingCycle(cycle) {
    billingCycle = cycle;
    
    document.querySelectorAll('.billing-btn').forEach(btn => btn.classList.remove('active'));

    if (cycle === 'semester') document.getElementById('btnCycleSemester').classList.add('active');

  // Sync user changes to active database array
  function updateUserDatabaseRecord(userNode) {
    localStorage.setItem('acadtrack_currentUser', JSON.stringify(userNode));
    let localUsers = [];
    try {
      const stored = localStorage.getItem('acadtrack_users');
      if (stored) localUsers = JSON.parse(stored);
    } catch (e) { localUsers = []; }

    const index = localUsers.findIndex(u => u.uid === userNode.uid);
    if (index !== -1) {
      localUsers[index] = userNode;
    } else {
      localUsers.push(userNode);
    }
    localStorage.setItem('acadtrack_users', JSON.stringify(localUsers));
  }
