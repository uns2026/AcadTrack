  function handleSlideResize() {
    const baseWidth = 1280;
    const baseHeight = 720;
    
    const scaleX = window.innerWidth / baseWidth;
    const scaleY = window.innerHeight / baseHeight;
    const scale = Math.min(scaleX, scaleY);

    const wrapper = document.getElementById('slide-wrapper');
    if (wrapper) {
      wrapper.style.transform = `translate(-50%, -50%) scale(${scale})`;
    }
  }

  function renderActiveSlide() {
    document.querySelectorAll('.slide-canvas').forEach(slide => {
      slide.classList.remove('active');
    });
    
    const activeSlide = document.getElementById(`slide-${currentSlideIndex}`);
    if (activeSlide) {
      activeSlide.classList.add('active');
    }

    // Sync global slide number trackers
    const trackers = document.querySelectorAll('#slideNumTracker');
    trackers.forEach(t => t.textContent = `${currentSlideIndex} / ${totalSlidesCount}`);
  }

  window.nextSlide = function() {
    if (currentSlideIndex < totalSlidesCount) {
      currentSlideIndex++;
      renderActiveSlide();
    }
  };

  window.prevSlide = function() {
    if (currentSlideIndex > 1) {
      currentSlideIndex--;
      renderActiveSlide();
    }
  };

      });
    } else {
      document.exitFullscreen();
