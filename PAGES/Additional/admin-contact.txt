<script id="inbox-portal-script">
  (function() {
    // Paste your Google Apps Script Web App Deployment URL here:
    const GOOGLE_APPS_SCRIPT_URL = "https://script.google.com/macros/s/AKfycbzcKp49sXa43kg0oBDkM0QfbxzoJ_KyZ2sXByqnaRn4Z7_SSfehQpTa9_2-FWjWv_0nSg/exec";

    let inboxDatabaseCache = [];
    let selectedMessageId = null;
    let searchQuery = '';

    function initInboxPortal() {
      // Auto-detect and inherit user language selection
      const initLang = localStorage.getItem('acadtrack-lang') || 'en';
      setLanguage(initLang);
      
      fetchRosterInboxMessages();
    }

    function setLanguage(lang) {
      localStorage.setItem('acadtrack-lang', lang);
      document.documentElement.lang = lang;
      const isRTL = lang === 'ar';
      const container = document.querySelector('.saas-container');
      if (container) container.dir = isRTL ? 'rtl' : 'ltr';

      document.querySelectorAll('[data-lang]').forEach(el => {
        el.style.display = el.getAttribute('data-lang') === lang ? '' : 'none';
      });
    }

    // Dynamic inbox fetch trigger (GET call)
    window.fetchRosterInboxMessages = function() {
      const refreshBtn = document.getElementById('refresh-btn');
      if (refreshBtn) refreshBtn.querySelector('i').className = "fa-solid fa-spinner";

      if (GOOGLE_APPS_SCRIPT_URL.includes("YOUR_DEPLOYED_URL")) {
        alert("Configuration Error: Please paste your active Google Apps Script web app URL inside the script block.");
        if (refreshBtn) refreshBtn.querySelector('i').className = "fa-solid fa-arrows-rotate";
        return;
      }

      fetch(GOOGLE_APPS_SCRIPT_URL + "?action=getMessages")
      .then(res => res.json())
      .then(res => {
        if (res && res.status === "success" && res.data) {
          inboxDatabaseCache = res.data;
          renderInboxRosterList();
        } else {
          alert("Error: " + (res.message || "Could not retrieve messages."));
        }
      })
      .catch(err => {
        alert("Network Error: Failed to connect to delivery gateway.");
      })
      .finally(() => {
        if (refreshBtn) refreshBtn.querySelector('i').className = "fa-solid fa-arrows-rotate";
      });
    };

    function renderInboxRosterList() {
      const listWrapper = document.getElementById('inbox-messages-list-wrapper');
      if (!listWrapper) return;

      const q = searchQuery.toLowerCase().trim();

      const filtered = inboxDatabaseCache.filter(m => {
        return !q ||
          (m.Name && m.Name.toLowerCase().includes(q)) ||
          (m.Subject && m.Subject.toLowerCase().includes(q)) ||
          (m.Message && m.Message.toLowerCase().includes(q)) ||
          (m.ID && m.ID.toLowerCase().includes(q));
      });

      if (filtered.length === 0) {
        listWrapper.innerHTML = `
          <div style="text-align:center; padding:40px 20px; color:var(--text-light);">
            <i class="fa-regular fa-envelope-open" style="font-size:24px; margin-bottom:8px; display:block;"></i>
            <span data-lang="en">No messages found</span><span data-lang="fr">Aucun message</span><span data-lang="ar">لا توجد أي رسائل واردة</span>
          </div>
        `;
        translateInboxPortalElements(listWrapper);
        return;
      }

      listWrapper.innerHTML = filtered.map(m => {
        const activeClass = m.ID === selectedMessageId ? 'active' : '';
        const isReplied = m.Status === 'Replied';
        
        const badgeHTML = isReplied ? 
          `<span class="status-badge replied"><span data-lang="en">Replied</span><span data-lang="fr">Répondu</span><span data-lang="ar">تم الرد</span></span>` : 
          `<span class="status-badge pending"><span data-lang="en">Pending</span><span data-lang="fr">En Attente</span><span data-lang="ar">معلقة</span></span>`;

        return `
          <div class="msg-item ${activeClass}" onclick="inspectActiveConversation('${m.ID}')">
            <div class="msg-item-top">
              <span class="msg-item-sender">${escapeHtml(m.Name)}</span>
              <span class="msg-item-time">${m.Timestamp ? m.Timestamp.split(',')[1] || m.Timestamp : ''}</span>
            </div>
            <div class="msg-item-sub">${escapeHtml(m.Subject)}</div>
            <div class="msg-item-body">${escapeHtml(m.Message)}</div>
            <div style="margin-top:6px; display:flex; justify-content:space-between; align-items:center;">
              ${badgeHTML}
              <span style="font-family:var(--font-mono); font-size:10px; color:var(--text-light); font-weight:700;">${m.ID}</span>
            </div>
          </div>
        `;
      }).join('');

      translateInboxPortalElements(listWrapper);
    }

    window.filterInboxRoster = function(val) {
      searchQuery = val;
      renderInboxRosterList();
    };

    // Displays full inquiry details and administrative reply editor
    window.inspectActiveConversation = function(id) {
      selectedMessageId = id;
      
      const record = inboxDatabaseCache.find(m => m.ID === id);
      if (!record) return;

      // Select row visually
      document.querySelectorAll('.msg-item').forEach(el => el.classList.remove('active'));
      renderInboxRosterList();

      document.getElementById('chat-empty-viewport').style.display = 'none';
      const viewport = document.getElementById('chat-active-viewport');
      viewport.style.display = 'flex';

      // Load header details
      document.getElementById('chat-sender-name').textContent = record.Name;
      document.getElementById('chat-sender-meta').textContent = `Role: ${record.Role.toUpperCase()} • Email: ${record.Email}`;

      const chatBody = document.getElementById('chat-scroller-body');
      if (!chatBody) return;

      let chatHTML = `
        <!-- Original Inquiry Bubble -->
        <div class="bubble-wrapper received">
          <div class="bubble-header"><span>${escapeHtml(record.Name)}</span> <span>·</span> <span style="font-family:var(--font-mono);">${record.Timestamp}</span></div>
          <div class="bubble-body">
            <div style="font-weight:800; font-size:13.5px; border-bottom:1px dashed #cbd5e1; padding-bottom:4px; margin-bottom:6px; color:#2563eb;">${escapeHtml(record.Subject)}</div>
            ${escapeHtml(record.Message)}
          </div>
        </div>
      `;

      // Append reply bubble if previously replied
      if (record.Status === 'Replied') {
        document.getElementById('chat-reply-input-bar').style.display = 'none';
        chatHTML += `
          <div class="bubble-wrapper sent">
            <div class="bubble-header"><span>Me</span> <span>·</span> <span style="font-family:var(--font-mono);">${record.ReplyTimestamp}</span></div>
            <div class="bubble-body">${escapeHtml(record.ReplyMessage)}</div>
          </div>
        `;
      } else {
        document.getElementById('chat-reply-input-bar').style.display = 'block';
        document.getElementById('reply-text-field').value = '';
      }

      chatBody.innerHTML = chatHTML;
      chatBody.scrollTop = chatBody.scrollHeight; // Scroll to bottom

      translateInboxPortalElements(viewport);
    };

    // Submits the admin's reply via fetch POST [1]
    window.submitOfficialAdminReply = function() {
      const replyText = document.getElementById('reply-text-field').value.trim();
      const submitBtn = document.getElementById('reply-submit-btn');
      const btnIcon = document.getElementById('reply-btn-icon');

      if (!replyText || !selectedMessageId) return;

      submitBtn.disabled = true;
      btnIcon.className = "fa-solid fa-spinner";

      const payload = {
        action: "replyMessage",
        messageId: selectedMessageId,
        replyText: replyText
      };

      // URL search parameters used for reliable compatibility with Google Apps Script
      const urlParams = new URLSearchParams(payload);

      fetch(GOOGLE_APPS_SCRIPT_URL, {
        method: "POST",
        body: urlParams,
        headers: {
          "Content-Type": "application/x-www-form-urlencoded"
        }
      })
      .then(res => res.json())
      .then(data => {
        if (data && data.status === "success") {
          // Re-fetch inbox, which will update states and lock the row to 'Replied'
          fetchRosterInboxMessages();
          
          // Re-render conversation frame with success details
          setTimeout(() => {
            inspectActiveConversation(selectedMessageId);
          }, 800);
        } else {
          alert("Error: " + (data.message || "Failed to submit administrative response."));
        }
      })
      .catch(err => {
        alert("Network Error: Failed to deliver administrative reply.");
      })
      .finally(() => {
        submitBtn.disabled = false;
        btnIcon.className = "fa-solid fa-paper-plane";
      });
    };

    function translateInboxPortalElements(container) {
      if (!container) return;
      const currentLang = localStorage.getItem('acadtrack-lang') || 'en';
      container.querySelectorAll('[data-lang]').forEach(el => {
        el.style.display = el.getAttribute('data-lang') === currentLang ? '' : 'none';
      });
    }

    function escapeHtml(text) {
      if (text == null) return '';
      const div = document.createElement('div');
      div.textContent = String(text);
      return div.innerHTML;
    }

    // Initialize page
    initInboxPortal();
  })();
</script>
