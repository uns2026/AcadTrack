<script>
(function () {

  const notifBtn = document.getElementById('notifBtnDesktop');
  const notifDropdown = document.getElementById('notifDropdownDesktop');
  const notifBadge = document.getElementById('notifCountBadge');
  const markAllReadBtn = document.getElementById('markAllReadBtn');

  const STORAGE_KEY = 'acadtrack_notifications_read';

  // Toggle dropdown
  function toggleDropdown(e) {
    e.stopPropagation();
    notifDropdown.classList.toggle('active');
  }

  // Apply read state UI
  function applyReadState() {

    const isRead = localStorage.getItem(STORAGE_KEY) === 'true';

    if (isRead) {

      // Hide badge
      if (notifBadge) {
        notifBadge.style.display = 'none';
      }

      // Fade notifications
      document.querySelectorAll('.notif-item').forEach(item => {
        item.classList.add('read');
      });

      // Disable button
      if (markAllReadBtn) {
        markAllReadBtn.textContent = 'All read';
        markAllReadBtn.classList.add('disabled');
      }

    } else {

      if (notifBadge) {
        notifBadge.style.display = 'inline-flex';
      }

      document.querySelectorAll('.notif-item').forEach(item => {
        item.classList.remove('read');
      });

      if (markAllReadBtn) {
        markAllReadBtn.textContent = 'Mark all read';
        markAllReadBtn.classList.remove('disabled');
      }
    }
  }

  // Mark notifications as read
  function markAllAsRead() {

    localStorage.setItem(STORAGE_KEY, 'true');

    applyReadState();
  }

  // Open / Close
  if (notifBtn) {
    notifBtn.addEventListener('click', toggleDropdown);
  }

  // Mark read click
  if (markAllReadBtn) {
    markAllReadBtn.addEventListener('click', markAllAsRead);
  }

  // Close outside click
  document.addEventListener('click', function (e) {

    if (
      notifDropdown &&
      !notifDropdown.contains(e.target) &&
      !notifBtn.contains(e.target)
    ) {
      notifDropdown.classList.remove('active');
    }
  });

  // Prevent dropdown click close
  notifDropdown.addEventListener('click', function (e) {
    e.stopPropagation();
  });

  // Restore saved state
  applyReadState();

})();
</script>
