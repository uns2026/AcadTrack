<script>
//<![CDATA[
  document.addEventListener('DOMContentLoaded', function() {
    // Run instant language alignment listener on load
    const initLang = localStorage.getItem('acadtrack-lang') || 'en';
    setLanguage(initLang);
  });

  function setLanguage(lang) {
    localStorage.setItem('acadtrack-lang', lang);
    document.documentElement.lang = lang;
    const isRTL = lang === 'ar';
    const section = document.getElementById('atPreviewSection');
    if (section) {
      section.dir = isRTL ? 'rtl' : 'ltr';
      if (isRTL) section.classList.add('lang-ar');
      else section.classList.remove('lang-ar');
    }

    document.querySelectorAll('.at-preview-section [data-lang]').forEach(el => {
      el.style.display = el.getAttribute('data-lang') === lang ? '' : 'none';
    });
  }

  // Interactive tab switcher logic [1]
  window.switchPreviewTab = function(role, btnEl) {
    // 1. Toggle Tab active styles
    document.querySelectorAll('.at-preview-tab').forEach(t => t.classList.remove('active'));
    if (btnEl) btnEl.classList.add('active');

    // 2. Hide all screenshots
    document.querySelectorAll('.browser-img').forEach(img => img.classList.remove('active'));
    
    // 3. Reveal active screenshot
    const activeImg = document.getElementById(`img-prev-${role}`);
    if (activeImg) activeImg.classList.add('active');

    // 4. Update Secure address bar URL [1]
    const urlField = document.getElementById('browserUrlField');
    if (urlField) {
      const pageMapping = {
        teacher: 'teacher-dashboard.html',
        student: 'student-dashboard.html',
        admin: 'dept-admin-dashboard.html'
      };
      urlField.textContent = `acadtrack.univ-batna.dz/${pageMapping[role] || 'dashboard.html'}`;
    }

    setLanguage(localStorage.getItem('acadtrack-lang') || 'en');
  };
//]]>
</script>
