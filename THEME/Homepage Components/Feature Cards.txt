//<![CDATA[
  document.addEventListener('DOMContentLoaded', function() {
    // Run instant language alignment listener on load
    const initLang = localStorage.getItem('acadtrack-lang') || 'en';
    setLanguage(initLang);

    // Initialize Touch/Grab-to-Scroll Horizontally for the three tracks (MOBILE ONLY)
    const tracks = document.querySelectorAll('.at-features-track');
    tracks.forEach(track => {
      let isDown = false;
      let startX;
      let scrollLeft;

      // Mouse drag start
      track.addEventListener('mousedown', (e) => {
        if (window.innerWidth >= 1024) return; // Disable drag/grab controls on Desktop
        isDown = true;
        track.classList.add('grabbing');
        startX = e.pageX - track.offsetLeft;
        scrollLeft = track.scrollLeft;
      });

      // Mouse drag exit/stop
      track.addEventListener('mouseleave', () => {
        isDown = false;
        track.classList.remove('grabbing');
      });
      track.addEventListener('mouseup', () => {
        isDown = false;
        track.classList.remove('grabbing');
      });

      // Active dragging move logic
      track.addEventListener('mousemove', (e) => {
        if (!isDown || window.innerWidth >= 1024) return; // Bypasses drag/grab on Desktop
        e.preventDefault();
        const x = e.pageX - track.offsetLeft;
        const walk = (x - startX) * 2; // Scroll-speed multiplier
        track.scrollLeft = scrollLeft - walk;
      });
    });

    // Auto-scroll each line's track to the 2nd card on mobile screen sizes [2]
    centerMobileActiveCards();
  });

  function centerMobileActiveCards() {
    if (window.innerWidth >= 1024) return; // Only process on mobile viewports [2]
    
    const tracks = document.querySelectorAll('.at-features-track');
    tracks.forEach(track => {
      const cards = track.querySelectorAll('.at-feat-card');
      if (cards.length >= 2) {
        const secondCard = cards[1]; // Index 1 is the 2nd card [2]
        
        // Calculate the symmetrical offset to align the 2nd card directly in the center [2]
        const scrollPosition = secondCard.offsetLeft - (track.clientWidth / 2) + (secondCard.clientWidth / 2);
        
        // Execute smooth scrolling positioning [2]
        track.scrollLeft = scrollPosition;
      }
    });
  }

  function setLanguage(lang) {
    localStorage.setItem('acadtrack-lang', lang);
    document.documentElement.lang = lang;
    const isRTL = lang === 'ar';
    const section = document.getElementById('atFeaturesSection');
    if (section) {
      section.dir = isRTL ? 'rtl' : 'ltr';
      if (isRTL) {
        section.classList.add('lang-ar');
        // Re-align on RTL language switch
        setTimeout(centerMobileActiveCards, 150);
      } else {
        section.classList.remove('lang-ar');
        setTimeout(centerMobileActiveCards, 150);
      }
    }

    document.querySelectorAll('.at-features-section [data-lang]').forEach(el => {
      el.style.display = el.getAttribute('data-lang') === lang ? '' : 'none';
    });
  }
