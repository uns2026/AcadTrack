//<![CDATA[
  document.addEventListener('DOMContentLoaded', function() {
    // Run instant language alignment listener on load
    const initLang = localStorage.getItem('acadtrack-lang') || 'en';
    setLanguage(initLang);

    // Auto-update copyright year dynamically [1]
    const year = new Date().getFullYear();
    const placeholders = ['current-year-placeholder', 'current-year-placeholder-fr', 'current-year-placeholder-ar'];
    placeholders.forEach(id => {
      const el = document.getElementById(id);
      if (el) el.textContent = year;
    });
  });

  function setLanguage(lang) {
    localStorage.setItem('acadtrack-lang', lang);
    document.documentElement.lang = lang;
    const isRTL = lang === 'ar';
    const section = document.getElementById('atFooterSection');
    if (section) {
      section.dir = isRTL ? 'rtl' : 'ltr';
      if (isRTL) section.classList.add('lang-ar');
      else section.classList.remove('lang-ar');
    }

    document.querySelectorAll('.at-footer-section [data-lang]').forEach(el => {
      el.style.display = el.getAttribute('data-lang') === lang ? '' : 'none';
    });
  }
//]]>
