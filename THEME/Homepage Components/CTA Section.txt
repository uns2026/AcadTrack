<script>
//<![CDATA[
  // Paste your Google Apps Script Web App Deployment URL here:
  const GOOGLE_APPS_SCRIPT_URL = "https://script.google.com/macros/s/AKfycbygbMOvDWKjxrgD_9VjtMAWjnlLWqd7AR-e8CNET0SX05bTGcf-qW29wTzQkVvxHNBb/exec";

  document.addEventListener('DOMContentLoaded', function() {
    const initLang = localStorage.getItem('acadtrack-lang') || 'en';
    setLanguage(initLang);
  });

  function setLanguage(lang) {
    localStorage.setItem('acadtrack-lang', lang);
    document.documentElement.lang = lang;
    const isRTL = lang === 'ar';
    const section = document.getElementById('atCtaSection');
    if (section) {
      section.dir = isRTL ? 'rtl' : 'ltr';
      if (isRTL) section.classList.add('lang-ar');
      else section.classList.remove('lang-ar');
    }

    document.querySelectorAll('.at-cta-section [data-lang], .waitlist-modal-box [data-lang]').forEach(el => {
      el.style.display = el.getAttribute('data-lang') === lang ? '' : 'none';
    });
  }

  window.openWaitlistModal = function() {
    const modal = document.getElementById('waitlist-modal');
    if (modal) {
      document.getElementById('waitlist-state-form').style.display = 'block';
      document.getElementById('waitlist-state-success').style.display = 'none';
      document.getElementById('waitlist-fields-form').reset();
      modal.classList.add('active');
    }
    setLanguage(localStorage.getItem('acadtrack-lang') || 'en');
  };

  window.closeWaitlistModal = function() {
    const modal = document.getElementById('waitlist-modal');
    if (modal) modal.classList.remove('active');
  };

  window.submitWaitlistSubscriber = function() {
    const first = document.getElementById('wl-first').value.trim();
    const last = document.getElementById('wl-last').value.trim();
    const email = document.getElementById('wl-email').value.trim();
    const phone = document.getElementById('wl-phone').value.trim();

    if (!first || !last || !email || !phone) return;

    // Ensure Apps Script URL is updated
    if (GOOGLE_APPS_SCRIPT_URL.includes("YOUR_DEPLOYED_URL")) {
      alert("Configuration Error: Please paste your active Google Apps Script web app URL inside the script block.");
      return;
    }

    const submitBtn = document.getElementById('wl-submit-btn');
    const btnIcon = document.getElementById('wl-btn-icon');
    const btnText = document.getElementById('wl-btn-text');
    const currentLang = localStorage.getItem('acadtrack-lang') || 'en';

    // Enter loading state
    if (submitBtn) submitBtn.disabled = true;
    if (btnIcon) btnIcon.style.display = "inline-block";
    if (btnText) {
      btnText.innerHTML = currentLang === 'ar' ? 'جاري الإرسال...' : (currentLang === 'fr' ? 'Envoi...' : 'Submitting...');
    }

    // Convert parameters to url-encoded format to bypass CORS blocks
    const payload = new URLSearchParams();
    payload.append('first', first);
    payload.append('last', last);
    payload.append('email', email);
    payload.append('phone', phone);

    fetch(GOOGLE_APPS_SCRIPT_URL, {
      method: "POST",
      body: payload,
      headers: {
        "Content-Type": "application/x-www-form-urlencoded"
      }
    })
    .then(res => res.json())
    .then(data => {
      if (data && data.status === "success") {
        // Transition modal window to show success state
        document.getElementById('waitlist-state-form').style.display = 'none';
        document.getElementById('waitlist-state-success').style.display = 'flex';
        setLanguage(currentLang);

        // Trigger floating success toast
        triggerSuccessToast(
          currentLang === 'ar' ? "تم التسجيل بنجاح" : "Registered Successfully", 
          currentLang === 'ar' ? "تم حجز مكانك في قائمة الانتظار وإرسال بريد لتأكيد ذلك." : "Your place on our early-bird waitlist is secured."
        );
      } else if (data && data.status === "duplicate") {
        const errMsg = {
          en: "This email address is already registered on our waitlist.",
          fr: "Cette adresse e-mail est déjà inscrite sur notre liste d'attente.",
          ar: "هذا البريد الإلكتروني مسجل بالفعل في قائمة الانتظار لدينا."
        };
        alert(errMsg[currentLang] || errMsg.en);
      } else {
        alert("Submission Failed: " + (data.message || "An unknown error occurred."));
      }
    })
    .catch(err => {
      const netError = {
        en: "Network Connection Error. Please verify your connection and try again.",
        fr: "Échec de la connexion. Veuillez vérifier votre réseau et réessayer.",
        ar: "خطأ في الاتصال بالشبكة. يرجى التحقق من اتصالك بالإنترنت وإعادة المحاولة."
      };
      alert(netError[currentLang] || netError.en);
    })
    .finally(() => {
      // Reset button states
      if (submitBtn) submitBtn.disabled = false;
      if (btnIcon) btnIcon.style.display = "none";
      if (btnText) {
        btnText.innerHTML = `
          <span data-lang="en">Submit</span><span data-lang="fr">Envoyer</span><span data-lang="ar">تأكيد التسجيل</span>
        `;
      }
      setLanguage(currentLang);
    });
  };

  function triggerSuccessToast(title, description) {
    const container = document.getElementById('profileToastContainer');
    if (!container) return;

    const currentLang = localStorage.getItem('acadtrack-lang') || 'en';
    const toastId = 'toast_' + Date.now();

    const toastHTML = `
      <div class="saas-toast-alert" id="${toastId}">
        <div class="saas-toast-icon">
          <i class="fa-solid fa-circle-check"></i>
        </div>
        <div class="saas-toast-body">
          <div class="saas-toast-title">${title}</div>
          <div class="saas-toast-desc">${description}</div>
        </div>
      </div>
    `;

    container.innerHTML += toastHTML;

    setTimeout(() => {
      const el = document.getElementById(toastId);
      if (el) el.classList.add('active');
    }, 50);

    setTimeout(() => {
      const el = document.getElementById(toastId);
      if (el) {
        el.classList.remove('active');
        setTimeout(() => el.remove(), 350);
      }
    }, 4500);
  }
//]]>
</script>
