<script>
//<![CDATA[
  document.addEventListener('DOMContentLoaded', function() {
    const initLang = localStorage.getItem('acadtrack-lang') || 'en';
    setLanguage(initLang);

    // Dynamic, high-performance IntersectionObserver ticking counters [1]
    const counters = document.querySelectorAll('.at-counter');
    const animationSpeed = 120; // Lower value = faster speed

    const observer = new IntersectionObserver(entries => {
      entries.forEach(entry => {
        if (entry.isIntersecting) {
          const counterEl = entry.target;
          const target = parseInt(counterEl.getAttribute('data-target') || '0');
          const suffix = counterEl.getAttribute('data-suffix') || '';
          
          let count = 0;
          const increment = Math.ceil(target / animationSpeed);

          const updateCount = () => {
            count += increment;
            if (count < target) {
              counterEl.textContent = count + suffix;
              setTimeout(updateCount, 15);
            } else {
              counterEl.textContent = target + suffix;
            }
          };

          updateCount();
          observer.unobserve(counterEl); // Unobserve once animated [1]
        }
      });
    }, { threshold: 0.3 }); // Ticks as soon as 30% of the element is visible

    counters.forEach(counter => observer.observe(counter));
  });

  function setLanguage(lang) {
    localStorage.setItem('acadtrack-lang', lang);
    document.documentElement.lang = lang;
    const isRTL = lang === 'ar';
    const section = document.getElementById('atStatsSection');
    if (section) {
      section.dir = isRTL ? 'rtl' : 'ltr';
      if (isRTL) section.classList.add('lang-ar');
      else section.classList.remove('lang-ar');
    }

    document.querySelectorAll('.at-stats-section [data-lang]').forEach(el => {
      el.style.display = el.getAttribute('data-lang') === lang ? '' : 'none';
    });
  }
//]]>
</script>
